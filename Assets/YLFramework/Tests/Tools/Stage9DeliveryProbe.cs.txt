        var stack = new Stack<IEnumerator>(); stack.Push(Run()); Exception failure = null;
        while (stack.Count != 0)
        {
            object current = null; bool more = false;
            try { more = stack.Peek().MoveNext(); if (more) current = stack.Peek().Current; }
            catch (Exception error) { failure = error; break; }
            if (!more) { (stack.Pop() as IDisposable)?.Dispose(); continue; }
            if (current is IEnumerator child && current is not CustomYieldInstruction) stack.Push(child);
            else yield return current;
        }
        while (stack.Count != 0)
            try { (stack.Pop() as IDisposable)?.Dispose(); } catch (Exception cleanup) { failure = failure == null ? cleanup : new AggregateException(failure, cleanup); }
        if (failure != null) { result["Failure"] = failure.ToString(); Debug.LogException(failure); }

        result["Errors"] = new JArray(errors); result["BusinessPassed"] = failure == null; result["ExitValidationPending"] = true; result["EndedUtc"] = DateTime.UtcNow.ToString("O");
        File.WriteAllText(output, result.ToString(Formatting.Indented));
        ended = true;
        Debug.Log(failure == null ? "STAGE9_PLAYER_PASS" : "STAGE9_PLAYER_FAILED");
        Application.Quit(failure == null ? 0 : 1);
    }

    private void Update()
    {
        if (output == null || ended || Time.realtimeSinceStartup - started < 150) return;
        ended = true;
        result["BusinessPassed"] = false; result["ExitValidationPending"] = true; result["Failure"] = "Owned consumer watchdog expired; no gate is claimed.";
        File.WriteAllText(output, result.ToString(Formatting.Indented)); Application.Quit(9);
    }
    private void Observe(string message, string trace, LogType type)
    {
        if (type == LogType.Error || type == LogType.Exception || type == LogType.Assert)
        {
            var text = type + ": " + message + "\n" + trace;
            errors.Add(text);
            // 保持监听至进程结束，不能让Application.Quit之后的清理错误漏过最终控制器。
            if (ended && output != null) File.AppendAllText(output + ".late-errors.log", text + "\n");
        }
    }

    private IEnumerator Click(RectTransform target)
    {
        Canvas.ForceUpdateCanvases(); yield return null;
        var canvas = target.GetComponentInParent<Canvas>().rootCanvas;
        var point = RectTransformUtility.WorldToScreenPoint(canvas.renderMode == RenderMode.ScreenSpaceOverlay ? null : canvas.worldCamera, target.TransformPoint(target.rect.center));
        Require(point.x > 0 && point.y > 0 && point.x < Screen.width && point.y < Screen.height, "UGUI target outside the rendered Player client.");
        var hits = new List<RaycastResult>();
        var data = new PointerEventData(EventSystem.current) { position = point, pressPosition = point,
            pointerId = -1, button = PointerEventData.InputButton.Left, eligibleForClick = true };
        EventSystem.current.RaycastAll(data, hits);
        Require(hits.Count != 0, "Actual Canvas raycaster found no UI at the rendered target.");
        data.pointerCurrentRaycast = hits[0]; data.pointerPressRaycast = hits[0];
        Step("ugui-input-" + (++clickIndex), new { ScreenX = point.x, ScreenY = point.y,
            Module = EventSystem.current.currentInputModule.GetType().FullName, Hits = hits.Select(x => x.gameObject.name).ToArray(), Canvas = canvas.renderMode.ToString(),
            Evidence = "Public EventSystem raycast and UGUI pointer handlers, not OS/native physical input" });
        ScreenCapture.CaptureScreenshot(Path.Combine(Path.GetDirectoryName(output), "before-input-" + clickIndex + ".png"));
        yield return new WaitForEndOfFrame();
        // 只向真实最上层命中对象派发，不跳过Mask命中去直接点击下层Button；不操作桌面全局输入。
        var hit = hits[0].gameObject;
        var pressed = ExecuteEvents.ExecuteHierarchy(hit, data, ExecuteEvents.pointerDownHandler);
        var click = ExecuteEvents.GetEventHandler<IPointerClickHandler>(hit);
        data.pointerPress = pressed;
        try
        {
            if (pressed) ExecuteEvents.Execute(pressed, data, ExecuteEvents.pointerUpHandler);
            if (click && click == pressed) ExecuteEvents.Execute(click, data, ExecuteEvents.pointerClickHandler);
        }
        finally { data.pointerPress = null; }
        yield return null;
    }

    private IEnumerator StopResources()
    {
        resources.ClearInstanceCaches();
        using (var stop = startup.StopAsync()) { yield return stop; stop.TakeResult(); }
        yield return null;
        var diagnostics = resources.GetDiagnosticsReport(); var content = resources.GetContentDiagnosticsSnapshot();
        Require(diagnostics.Summary.LeaseCount == 0 && diagnostics.Assets.Count == 0 && diagnostics.Bundles.Count == 0 && diagnostics.InstanceCaches.Count == 0 && diagnostics.Scenes.Count == 0,
            "All UI/asset/instance resource responsibilities must drain.");
        Require(diagnostics.Summary.NativeBundleLoadCount == diagnostics.Summary.NativeBundleUnloadCount && content.WorkCount == 0 && content.LiveInputs == 0 && content.ProcessCngKeyCount == 0 && content.ProcessCngProviderCount == 0,
            "Real native Bundle and CNG owners must release.");
        Require(ResourceSystem.Current == null && CoreResourceAccess.Provider == null && !AssetBundle.GetAllLoadedAssetBundles().Any(), "Shutdown must release authority, Core binding and native Bundle instances.");
        Step("resource-stop-clean", new { Diagnostics = diagnostics, Content = content });
    }

    }
    private IEnumerator LoggerCoexistence()
    {
        const string managerName = "XFGameFramework.LoggerSystem.LoggerManager";
        const string handlerName = "XFGameFramework.LoggerSystem.LoggerHandler";
        var assembly = AppDomain.CurrentDomain.GetAssemblies().Single(a => a.GetName().Name == "XFGameFramework.LoggerSystem");
        var type = assembly.GetType(managerName, true);
        var manager = FindObjectsOfType<MonoBehaviour>().Single(x => x.GetType() == type);
        Require(Debug.unityLogger.logHandler.GetType().FullName == handlerName, "The actual optional Logger must install its own handler.");
        var property = type.GetProperty("OnLogEvent");
        var old = (Action<LogType, string>)property.GetValue(null);
        var observed = 0;
        Action<LogType, string> witness = (kind, text) => { if (text == "STAGE9_LOGGER_FILE_WITNESS") observed++; };
        property.SetValue(null, old + witness);
        try { Debug.Log("STAGE9_LOGGER_FILE_WITNESS"); }
        finally { property.SetValue(null, old); }
        Require(observed == 1, "Optional Logger public event dispatch.");
        var firstPath = (string)type.GetProperty("LogFilePath").GetValue(null);
        Destroy(manager.gameObject); yield return null;
        Require(Debug.unityLogger.logHandler.GetType().FullName == "UnityEngine.DebugLogHandler", "Normal Logger owner destruction must return the native handler.");
        var native = Debug.unityLogger.logHandler;
        string firstText;
        using (var file = File.Open(firstPath, FileMode.Open, FileAccess.Read, FileShare.None))
        using (var reader = new StreamReader(file)) firstText = reader.ReadToEnd();
        Require(firstText.Contains("STAGE9_LOGGER_FILE_WITNESS"), "Optional Logger writer did not persist or release its file.");
        File.WriteAllText(Path.Combine(Path.GetDirectoryName(output), "optional-logger-first.log"), firstText);
        // 使用公开属性显式重建，验证已释放的static writer不会被下一实例复用。
        Require(type.GetProperty("FilterLogType").GetValue(null) != null, "Optional Logger explicit reinitialization.");
        manager = FindObjectsOfType<MonoBehaviour>().Single(x => x.GetType() == type);
        Require(Debug.unityLogger.logHandler.GetType().FullName == handlerName, "Reinitialized Logger must install a new handler.");
        Debug.Log("STAGE9_LOGGER_SECOND_WRITER_WITNESS");
        var secondPath = (string)type.GetProperty("LogFilePath").GetValue(null);
        // 非所有者的后续handler接管必须保持，测试只归还自己安装的包装。
        var foreign = new Stage9OwnedLogHandler(native);
        Debug.unityLogger.logHandler = foreign;
        try
        {
            Destroy(manager.gameObject); yield return null;
            Require(ReferenceEquals(Debug.unityLogger.logHandler, foreign), "Logger destruction must not overwrite a later handler owner.");
            Debug.Log("STAGE9_LOGGER_FOREIGN_WITNESS"); Require(foreign.Witnesses == 1, "Later handler ownership not usable.");
        }
        finally { if (ReferenceEquals(Debug.unityLogger.logHandler, foreign)) Debug.unityLogger.logHandler = native; }
        string secondText;
        using (var file = File.Open(secondPath, FileMode.Open, FileAccess.Read, FileShare.None))
        using (var reader = new StreamReader(file)) secondText = reader.ReadToEnd();
        Require(secondText.Contains("STAGE9_LOGGER_SECOND_WRITER_WITNESS"), "New Logger reused a disposed writer or failed to close it.");
        File.WriteAllText(Path.Combine(Path.GetDirectoryName(output), "optional-logger-second.log"), secondText);
        Require(!FindObjectsOfType<MonoBehaviour>().Any(x => x.GetType() == type), "Optional Logger instance must not respawn during normal cleanup.");
        Step("optional-logger-owner-writer-coexistence", new { EventWitnesses = observed, LaterOwnerWitnesses = foreign.Witnesses,
            HandlerReturned = Debug.unityLogger.logHandler.GetType().FullName, FileExclusiveAfterBothDestructions = true,
            Scope = "Separate MIT LoggerSystem 1.0.1 patched artifact; primary product-only eight-case evidence predates installation" });
    }

public sealed class Stage9DeliveryKeys : IResourceContentKeyProvider
{
    private readonly bool missing;
    public Stage9DeliveryKeys(bool missing = false) { this.missing = missing; }
    public bool TryCopyKey(string projectId, string keyId, byte[] destination)
    {
        if (missing || projectId != "stage9-encrypted" || keyId != "stage9-nonproduction") return false;
        // 固定的公开非生产夹具输入；不输出、序列化或交付任何生产秘密。
        using var hash = SHA256.Create(); var key = hash.ComputeHash(Encoding.ASCII.GetBytes("Stage9 public nonproduction acceptance fixture only"));
        try { Buffer.BlockCopy(key, 0, destination, 0, key.Length); return true; } finally { Array.Clear(key, 0, key.Length); }
    }
