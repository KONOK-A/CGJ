using System;
using System.IO;
using System.Linq;
using System.Security.Cryptography;
using Newtonsoft.Json;
using Newtonsoft.Json.Linq;
using UnityEditor;
using UnityEditor.Compilation;
using UnityEngine;

namespace YLFramework.Tests.DomainReception.Editor
{
    public static class DomainReceptionExcel
    {
        private const string Project = "G:/Unity/MC26-20261020T173208Z/Receiver";
        private static string Step()
        {
            if (!Application.isBatchMode || Application.unityVersion != "2022.3.62f3c1" ||
                Path.GetFullPath(Path.GetDirectoryName(Application.dataPath)).Replace('\\', '/') != Project)
                throw new InvalidOperationException("Owned independent receiver only; never call this in the main Editor.");
            var owner = JObject.Parse(File.ReadAllText(Project + "/domain-reception-owner.json"));
            if ((string)owner["purpose"] != "MC26-C-Reception" || (string)owner["cCommit"] != "aadc11b5099b76f16667aa9818ca927104cd12be")
                throw new InvalidOperationException("Wrong optional reception owner.");
            var args = Environment.GetCommandLineArgs(); int index = Array.IndexOf(args, "-domainExcelStep");
            if (index < 0 || index + 1 == args.Length) throw new InvalidOperationException("Missing optional step.");
            string step = Path.GetFullPath(args[index + 1]).Replace('\\', '/');
            if (!step.StartsWith("G:/Unity/MC26-20261020T173208Z/Reception/Runs/", StringComparison.Ordinal) ||
                !File.Exists(step + "/request.json") || File.Exists(step + "/optional-editor.json"))
                throw new InvalidOperationException("New receipt-owned optional step required.");
            return step;
        }

        public static void Enable()
        {
            string step = Step(), defines = PlayerSettings.GetScriptingDefineSymbolsForGroup(BuildTargetGroup.Standalone);
            if (defines.Split(';').Contains("YL_DOMAIN_EXCEL")) throw new InvalidOperationException("Optional define already enabled.");
            PlayerSettings.SetScriptingDefineSymbolsForGroup(BuildTargetGroup.Standalone, defines + (defines.Length == 0 ? "" : ";") + "YL_DOMAIN_EXCEL");
            File.WriteAllText(step + "/optional-editor.json", JsonConvert.SerializeObject(new { passed = true, previousDefines = defines,
                enabled = true, compilationClaimed = false, pid = System.Diagnostics.Process.GetCurrentProcess().Id }, Formatting.Indented));
        }

        public static void Audit()
        {
            string step = Step();
            if (EditorUtility.scriptCompilationFailed || !PlayerSettings.GetScriptingDefineSymbolsForGroup(BuildTargetGroup.Standalone).Split(';').Contains("YL_DOMAIN_EXCEL"))
                throw new InvalidOperationException("Actual optional compile/define failed.");
            string[] expected = { "YLFramework.Domain.Excel.Editor", "YLFramework.Domain.Excel.Tests", "YLFramework.Excel.Editor" };
            var editor = CompilationPipeline.GetAssemblies(AssembliesType.Editor);
            if (expected.Any(n => editor.All(a => a.name != n))) throw new InvalidOperationException("Missing real optional assembly.");
            var player = CompilationPipeline.GetAssemblies(AssembliesType.Player);
            if (player.Any(a => expected.Contains(a.name) || a.compiledAssemblyReferences.Any(p => Path.GetFileNameWithoutExtension(p).Contains("MiniExcel"))))
                throw new InvalidOperationException("Optional Editor backend entered Player compilation.");
            const string dll = "Assets/YLFramework/Integrations/Excel/Editor/Plugins/MiniExcel.dll";
            var plugin = AssetImporter.GetAtPath(dll) as PluginImporter;
            if (!plugin || !plugin.GetCompatibleWithEditor() || plugin.GetCompatibleWithAnyPlatform() || plugin.GetCompatibleWithPlatform(BuildTarget.StandaloneWindows64))
                throw new InvalidOperationException("MiniExcel is not Editor-only.");
            string hash; using (var sha = SHA256.Create()) using (var stream = File.OpenRead(dll)) hash = BitConverter.ToString(sha.ComputeHash(stream)).Replace("-", "").ToLowerInvariant();
            if (hash != "0a49b66224bef9fdcec12800e4660267a993b1ece87727d9fe15c7dabf58ba1e" || new FileInfo(dll).Length != 200192) throw new InvalidOperationException("MiniExcel pin changed.");
            File.WriteAllText(step + "/optional-editor.json", JsonConvert.SerializeObject(new { passed = true, actualCompiled = true,
                pid = System.Diagnostics.Process.GetCurrentProcess().Id, unity = Application.unityVersion, miniExcelSha256 = hash,
                validationPin = JObject.Parse(File.ReadAllText("Assets/YLFramework/Integrations/DomainExcel/Editor/Dependencies.json")),
                actualBackendProductTip = "5dce4f3e39e2f4bd11602e65cabace5f56108cf8",
                editor = editor.Where(a => expected.Contains(a.name)).Select(a => new { a.name, a.sourceFiles, a.defines, a.compiledAssemblyReferences }),
                packages = UnityEditor.PackageManager.PackageInfo.GetAllRegisteredPackages().Select(p => new { p.name, p.version, source = p.source.ToString() }) }, Formatting.Indented));
        }
    }
}
