using System;
using System.IO;
using System.Reflection;
using Newtonsoft.Json;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEditor.TestTools.TestRunner.Api;
using UnityEngine;

namespace YLFramework.Tests.EquipmentAcceptance
{
    [InitializeOnLoad]
    public static class EquipmentReceiverExcel
    {
        private const string Key = "YLFramework.Equipment.Receiver.Excel";
        private static int stable;
        private static TestRunnerApi api;
        static EquipmentReceiverExcel()
        {
            if (!string.IsNullOrEmpty(SessionState.GetString(Key, "")))
            {
                if (SessionState.GetBool(Key + ".Finished", false)) EditorApplication.update += Finish;
                else EditorApplication.update += Begin;
            }
        }
        public static void Run()
        {
            if (!Application.isBatchMode || !File.Exists("equipment-receiver.json")) throw new InvalidOperationException("Only the owned receiver may run this gate.");
            var identity = Newtonsoft.Json.Linq.JObject.Parse(File.ReadAllText("equipment-receiver.json"));
            if ((string)identity["purpose"] != "Equipment-independent-receiver" || Path.GetFullPath((string)identity["project"]) != Path.GetFullPath(".")) throw new InvalidOperationException("Receiver identity differs.");
            var scene = EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);
            if (!EditorSceneManager.SaveScene(scene, "Assets/EquipmentHarness/Empty.unity")) throw new IOException("Cannot save the owned empty scene.");
            var archive = Path.Combine(Directory.GetParent(Path.GetFullPath(".")).FullName, "Gates/Excel");
            if (Directory.Exists(archive)) throw new IOException("Preserve prior Excel evidence.");
            EquipmentSettings.Capture("Temp/EquipmentAcceptance/Excel");
            SessionState.SetString(Key, archive);
            EquipmentSettings.EnableExcel();
            EditorApplication.update -= Begin; EditorApplication.update += Begin;
        }
        private static void Begin()
        {
            if (EditorApplication.isCompiling || EditorApplication.isUpdating) { stable = 0; return; }
#if YL_DOMAIN_EXCEL
            if (++stable < 8) return;
            EditorApplication.update -= Begin;
            api = ScriptableObject.CreateInstance<TestRunnerApi>();
            api.RegisterCallbacks(new Results());
            api.Execute(new ExecutionSettings(new Filter { testMode = TestMode.EditMode, groupNames = new[] { "^YLFramework[.]Domain[.]Excel[.]Tests[.]" } }));
#endif
        }
        private sealed class Results : ICallbacks
        {
            public void RunStarted(ITestAdaptor tests) { }
            public void TestStarted(ITestAdaptor test) { }
            public void TestFinished(ITestResultAdaptor result) { }
            public void RunFinished(ITestResultAdaptor result)
            {
                Directory.CreateDirectory("Temp/EquipmentAcceptance/Excel");
                File.WriteAllText("Temp/EquipmentAcceptance/Excel/results.xml", result.ToXml().OuterXml);
                File.WriteAllText("Temp/EquipmentAcceptance/Excel/summary.json", JsonConvert.SerializeObject(new { result.ResultState, result.PassCount, result.FailCount, result.SkipCount, result.InconclusiveCount, result.Duration }));
                SessionState.SetBool(Key + ".Passed", result.PassCount == 15 && result.FailCount == 0 && result.SkipCount == 0 && result.InconclusiveCount == 0);
                SessionState.SetBool(Key + ".Finished", true);
                EditorApplication.update -= Finish; EditorApplication.update += Finish;
            }
        }
        private static void Finish()
        {
            if (EditorApplication.isCompiling || EditorApplication.isUpdating || (bool)typeof(TestRunnerApi).GetMethod("IsRunActive", BindingFlags.NonPublic | BindingFlags.Static).Invoke(null, null)) return;
            EditorApplication.update -= Finish;
            var passed = SessionState.GetBool(Key + ".Passed", false);
            string error = null;
            try { EquipmentSettings.Restore(); }
            catch (Exception failure) { passed = false; error = failure.ToString(); Debug.LogException(failure); }
            var gate = Path.GetFullPath("Temp/EquipmentAcceptance/Excel");
            File.WriteAllText(gate + "/gate-complete.json", JsonConvert.SerializeObject(new { passed, error, utc = DateTime.UtcNow }));
            var archive = SessionState.GetString(Key, ""); Directory.CreateDirectory(archive);
            foreach (var file in Directory.GetFiles(gate)) File.Copy(file, Path.Combine(archive, Path.GetFileName(file)));
            SessionState.EraseString(Key); SessionState.EraseBool(Key + ".Finished"); SessionState.EraseBool(Key + ".Passed");
            EditorApplication.Exit(passed ? 0 : 1);
        }
    }
}
