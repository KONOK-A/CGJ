namespace YLFramework.Tests.Resource
{
    public static class HybridCLRFixtureHost
    {
        public static void Run(string root, string publication, string readySuffix)
        {
            var directory = System.IO.Path.GetFullPath(System.IO.Path.Combine(root, publication));
            if (!System.IO.File.Exists(directory + "/release.json")) throw new System.IO.FileNotFoundException("Sign the owned candidate release first.");
            var ready = root + "/" + publication + (readySuffix.Length == 0 ? "" : "-" + readySuffix) + "-ready.json";
            if (System.IO.File.Exists(ready)) throw new System.IO.IOException("Preserve prior server evidence; do not replace a ready receipt.");
            var files = new System.Collections.Generic.Dictionary<string, byte[]>(System.StringComparer.Ordinal);
            foreach (var file in System.IO.Directory.GetFiles(directory, "*", System.IO.SearchOption.AllDirectories))
                files.Add("/" + file.Substring(directory.Length + 1).Replace((char)92, (char)47), System.IO.File.ReadAllBytes(file));
            using (var certificate = System.Security.Cryptography.X509Certificates.X509CertificateLoader.LoadPkcs12FromFile(root + "/TestCredentials/local-only.pfx", "hclr-local-fixture",
                System.Security.Cryptography.X509Certificates.X509KeyStorageFlags.DefaultKeySet))
            using (var server = new ResourceTlsServer(certificate, request => files.TryGetValue(request.Path, out var body)
                ? new ResourceTlsServer.Response { Body = body, Headers = { ["ETag"] = "\"hclr-stable\"" } } : new ResourceTlsServer.Response { StatusCode = 404 }))
            using (var rsa = certificate.GetRSAPublicKey())
            using (var stopping = new System.IO.FileSystemWatcher(root, System.IO.Path.GetFileName(ready) + ".stop"))
            {
                var stopPath = ready + ".stop";
                if (System.IO.File.Exists(stopPath)) throw new System.IO.IOException("Preserve the previous fixture stop signal; choose a new ready suffix.");
                stopping.EnableRaisingEvents = true;
                var key = rsa.ExportParameters(false);
                // Values are public fixture trust material only; no private key is emitted.
                var json = "{\"endpoint\":\"" + server.BaseUri.AbsoluteUri + "\",\"modulus\":\"" + System.Convert.ToBase64String(key.Modulus) +
                    "\",\"exponent\":\"" + System.Convert.ToBase64String(key.Exponent) + "\",\"certificate\":\"" + System.Convert.ToBase64String(certificate.RawData) + "\"}";
                System.IO.File.WriteAllText(ready, json);
                System.Console.WriteLine("HybridCLR loopback HTTPS fixture ready: " + server.BaseUri.AbsoluteUri);
                if (!System.IO.File.Exists(stopPath)) stopping.WaitForChanged(System.IO.WatcherChangeTypes.Created | System.IO.WatcherChangeTypes.Renamed);
                System.Console.WriteLine("Accepted requests=" + server.RequestCount + "; fixture errors=" + server.Errors.Count);
            }
        }
    }
}
