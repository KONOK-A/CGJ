using System;
using System.Collections;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Reflection;
using System.Text;
using System.Threading.Tasks;
using Newtonsoft.Json;
using Newtonsoft.Json.Linq;
using UnityEngine;
using YLFramework.Integrations.HybridCLR;
using YLFramework.Resource;

namespace YLFramework.Tests.HybridCLRReception
{
    public sealed class HybridCLRProjectSubsetProbe : MonoBehaviour
    {
        private const string ExpectedUnityVersion = "__YL_HYBRIDCLR_EXPECTED_UNITY_VERSION__";
        private const string ExpectedUnityRevision = "__YL_HYBRIDCLR_EXPECTED_UNITY_REVISION__";
        private const string ExpectedRoot = "__YL_HYBRIDCLR_EXPECTED_ROOT__";
        private const string BoundOwnerSha256 = "__YL_HYBRIDCLR_OWNER_SHA256__";
        private const string SupportName = "YLFramework.HybridCLR.Sample.Support";
        private const string EntryName = "YLFramework.HybridCLR.Sample.Entry";
        public string compatibility = "__YL_HYBRIDCLR_SUBSET_COMPATIBILITY__";
        public string playerDirectory = "__YL_HYBRIDCLR_SUBSET_PLAYER__";
        public string cacheDirectory = "__YL_HYBRIDCLR_SUBSET_CACHE__";
        private readonly JObject _evidence = new JObject
        {
            ["action"] = "ProjectSubsets",
            ["simulation"] = JValue.CreateNull(),
            ["loads"] = new JArray(),
            ["sharedSupportAssembly"] = JValue.CreateNull(),
            ["barrierAReleased"] = false,
            ["barrierBReleased"] = false,
            ["stopped"] = false,
            ["asyncOutput"] = JValue.CreateNull()
        };
        private string _root;
        private string _run;
        private ResourceStartup _startup;
        private HybridCLRLoader _loaderA;
        private HybridCLRLoader _loaderB;
        private HybridCLRLoadResult _loadedA;
        private HybridCLRLoadResult _loadedB;

        private static string Argument(string name)
        {
            var args = Environment.GetCommandLineArgs();
            var index = Array.IndexOf(args, name);
            if (index < 0 || index + 1 == args.Length || index != Array.LastIndexOf(args, name))
            {
                throw new ArgumentException("One explicit " + name + " is required.");
            }

            return args[index + 1];
        }

        private IEnumerator Start()
        {
            Application.runInBackground = true;
            var failures = new List<Exception>();
            yield return Observe(Run(), failures);
            // Release both manifest barriers and the owned Resource startup even after a load/assertion failure.
            yield return Observe(Close(failures), failures);
            _evidence["passed"] = failures.Count == 0;
            if (failures.Count != 0)
            {
                _evidence["error"] = string.Join("\n", failures.Select(e => e.ToString()));
            }

            try
            {
                if (string.IsNullOrEmpty(_root) || string.IsNullOrEmpty(_run))
                {
                    throw new InvalidOperationException("Missing receipt destination.");
                }

                using var file = new FileStream(Path.Combine(_root, _run + ".json"), FileMode.CreateNew, FileAccess.Write, FileShare.Read);
                var bytes = Encoding.UTF8.GetBytes(_evidence.ToString(Formatting.Indented));
                file.Write(bytes, 0, bytes.Length);
                file.Flush(true);
                Debug.Log("HYBRIDCLR_PROJECT_SUBSETS " + _evidence.ToString(Formatting.None));
            }
            catch (Exception error)
            {
                failures.Add(error);
                Debug.LogException(error);
            }
            Application.Quit(failures.Count == 0 ? 0 : 21);
        }

        private static IEnumerator Observe(IEnumerator routine, List<Exception> failures)
        {
            var stack = new Stack<IEnumerator>();
            stack.Push(routine);
            var failed = false;
            while (stack.Count != 0 && !failed)
            {
                var next = false;
                object current = null;
                try
                {
                    next = stack.Peek().MoveNext();
                    if (next)
                    {
                        current = stack.Peek().Current;
                    }
                }
                catch (Exception error)
                {
                    failures.Add(error);
                    Debug.LogException(error);
                    failed = true;
                }
                if (failed)
                {
                    break;
                }

                if (!next)
                {
                    try
                    {
                        (stack.Pop() as IDisposable)?.Dispose();
                    }
                    catch (Exception error)
                    {
                        failures.Add(error);
                        Debug.LogException(error);
                        failed = true;
                    }
                    continue;
                }
                if (current is IEnumerator nested && !(current is CustomYieldInstruction))
                {
                    stack.Push(nested);
                    continue;
                }
                yield return current;
            }
            while (stack.Count != 0)
            {
                try
                {
                    (stack.Pop() as IDisposable)?.Dispose();
                }
                catch (Exception error)
                {
                    failures.Add(error);
                    Debug.LogException(error);
                }
            }
        }

        private IEnumerator Run()
        {
            _root = Path.GetFullPath(Argument("-hclr-root"));
            var requestedRun = Argument("-hclr-run");
            Require(System.Text.RegularExpressions.Regex.IsMatch(requestedRun, "^[A-Za-z0-9-]+$") &&
                requestedRun != "project-subsets-base" && requestedRun != "project-subsets-patch", "Use a new explicit Player run id.");
            _run = requestedRun;
            Require(!File.Exists(Path.Combine(_root, _run + ".json")) && !Directory.Exists(Path.Combine(_root, _run + ".json")), "Preserve the earlier Player receipt.");
            Require(Argument("-hclr-action") == "ProjectSubsets", "This scene accepts only ProjectSubsets.");
            Require(Application.platform == RuntimePlatform.WindowsPlayer && IntPtr.Size == 8 && Application.unityVersion == ExpectedUnityVersion &&
                Path.GetFullPath(_root).Equals(Path.GetFullPath(ExpectedRoot), StringComparison.OrdinalIgnoreCase),
                "The probe requires its actual Windows64 IL2CPP Player and exact Unity version.");
#if !ENABLE_IL2CPP
            throw new PlatformNotSupportedException("The project-subset probe must run in an IL2CPP Player.");
#endif
            var basePath = Path.Combine(_root, "project-subsets-base.json");
            Require(Path.GetFullPath(Argument("-hclr-base-receipt")) == Path.GetFullPath(basePath), "Wrong explicit base receipt.");
            var baseline = JObject.Parse(File.ReadAllText(basePath));
            var patch = JObject.Parse(File.ReadAllText(Path.Combine(_root, "project-subsets-patch.json")));
            var conflict = JObject.Parse(File.ReadAllText(Path.Combine(_root, "project-subsets-conflict.json")));
            if (BoundOwnerSha256.Length != 0)
            {
                var ownerPath = Path.Combine(_root, "owner.json");
                var owner = JObject.Parse(File.ReadAllText(ownerPath));
                Require(HybridCLRManifest.Hash(File.ReadAllBytes(ownerPath)) == BoundOwnerSha256 && !owner.Value<bool>("simulation"), "Wrong immutable subset owner.");
                foreach (var receipt in new[] { baseline, patch, conflict })
                {
                    Require(receipt.Value<string>("UnityVersion") == ExpectedUnityVersion && receipt.Value<string>("UnityRevision") == ExpectedUnityRevision &&
                        receipt.Value<string>("Receiver") == owner.Value<string>("receiver") && receipt.Value<string>("OwnerSha256") == BoundOwnerSha256 &&
                        receipt.Value<int>("SubsetConsumerVersion") == 2 &&
                        Path.GetFullPath(receipt.Value<string>("Baseline")) == Path.GetFullPath(owner.Value<string>("subsetBaseline")) &&
                        Path.GetFullPath(playerDirectory) == Path.GetFullPath(owner.Value<string>("subsetPlayerRoot")) &&
                        Path.GetFullPath(cacheDirectory) == Path.GetFullPath(owner.Value<string>("subsetCacheRoot")) && compatibility == owner.Value<string>("subsetCompatibilityId"),
                        "Cross-engine or foreign subset baseline/paths.");
                }
                if (owner["nativeSourceManifest"] != null)
                {
                    Require(ExpectedUnityVersion == "6000.3.27f1" && new[] { baseline, patch, conflict }.All(receipt =>
                        receipt.Value<string>("NativeVariant") == "ParameterOwner" &&
                        receipt.Value<string>("NativeSourceManifestSha256") == owner.Value<string>("nativeSourceManifestSha256")), "Wrong maintained native subset source binding.");
                    _evidence["nativeVariant"] = "ParameterOwner";
                    _evidence["nativeSourceManifestSha256"] = owner.Value<string>("nativeSourceManifestSha256");
                }
                else
                {
                    _evidence["nativeVariant"] = "OfficialSdkLocal";
                }
            }
            Require(baseline.Value<bool>("passed") && patch.Value<bool>("passed") &&
                baseline.Value<string>("CompatibilityId") == compatibility && patch.Value<string>("CompatibilityId") == compatibility &&
                baseline.Value<bool>("Development") == Debug.isDebugBuild && Debug.isDebugBuild && patch.Value<bool>("Development"),
                "The explicit base/patch identity or actual Development flag does not match.");
            Require(Path.GetFullPath(baseline.Value<string>("PlayerDirectory")) == Path.GetFullPath(playerDirectory) &&
                Path.GetFullPath(Path.GetDirectoryName(Application.dataPath)) == Path.GetFullPath(playerDirectory) &&
                Path.GetFullPath(baseline.Value<string>("RuntimeCacheDirectory")) == Path.GetFullPath(cacheDirectory) &&
                baseline.Value<string>("PlayerFingerprint") == patch.Value<string>("PlayerFingerprint") &&
                baseline.Value<string>("BaselineFingerprint") == patch.Value<string>("BaselineFingerprint") &&
                baseline.Value<string>("BaselineSha256") == patch.Value<string>("BaselineSha256"), "Wrong Player, cache or frozen base binding.");
            Require(JToken.DeepEquals(baseline["ReleaseA"], patch["ReleaseA"]) && JToken.DeepEquals(baseline["SharedSupport"], patch["SharedSupport"]),
                "Patch evidence changed A's published release or shared Support descriptor.");
            Require(conflict.Value<bool>("passed") && conflict.Value<bool>("FixtureOnly") && conflict.Value<string>("CompatibilityId") == compatibility &&
                conflict.Value<string>("PlayerFingerprint") == baseline.Value<string>("PlayerFingerprint") &&
                conflict.Value<string>("BaselineFingerprint") == baseline.Value<string>("BaselineFingerprint") &&
                JToken.DeepEquals(conflict["OriginalSupport"], baseline["SharedSupport"]) &&
                conflict["ChangedSupport"].Value<string>("Identity") == baseline["SharedSupport"].Value<string>("Identity") &&
                conflict["ChangedSupport"].Value<string>("Mvid") != baseline["SharedSupport"].Value<string>("Mvid"), "Foreign or non-conflicting shared Support fixture.");
            var releaseA = (JObject)baseline["ReleaseA"];
            var releaseB = (JObject)patch["ReleaseB"];
            Require(releaseA.Value<string>("ProjectId") == "hclr-sub-a" && releaseB.Value<string>("ProjectId") == "hclr-sub-b", "Wrong project subset ids.");
            var snapshotA = ResourceLocalSnapshot.Open(releaseA.Value<string>("DirectoryPath"), "hclr-sub-a", releaseA.Value<string>("CatalogDigest"), "StandaloneWindows64", compatibility);
            var snapshotB = ResourceLocalSnapshot.Open(releaseB.Value<string>("DirectoryPath"), "hclr-sub-b", releaseB.Value<string>("CatalogDigest"), "StandaloneWindows64", compatibility);
            var conflictB = ResourceLocalSnapshot.Open(conflict["ReleaseB"].Value<string>("DirectoryPath"), "hclr-sub-b", conflict["ReleaseB"].Value<string>("CatalogDigest"), "StandaloneWindows64", compatibility);
            Require(!Directory.Exists(cacheDirectory) && !File.Exists(cacheDirectory) && !File.Exists(cacheDirectory + ".meta"), "Never adopt an earlier runtime cache.");
            _evidence["unityVersion"] = Application.unityVersion;
            _evidence["development"] = Debug.isDebugBuild;
            _evidence["source"] = "explicit-trusted-local-snapshots";
            _evidence["releaseA"] = releaseA.DeepClone();
            _evidence["releaseB"] = releaseB.DeepClone();
            _evidence["cacheDirectory"] = cacheDirectory;
            var conflictStore = new ArtifactStore(Path.Combine(cacheDirectory, "conflict"), new[]
            {
                new ResourceProjectMount("hclr-sub-a", "StandaloneWindows64", compatibility, snapshotA),
                new ResourceProjectMount("hclr-sub-b", "StandaloneWindows64", compatibility, conflictB)
            });
            yield return StartStore(conflictStore);

            using (var foreign = new HybridCLRLoader(_startup.Resources, compatibility))
            {
                var denied = foreign.LoadAsync(new ResourceAddress("hclr-sub-foreign", HybridCLRManifest.ManifestPath));
                yield return denied;
                Require(denied.Exception != null && denied.Result == null && !denied.RequiresRestart, "An unmounted foreign project must not borrow either loaded project or reach native loading.");
                _evidence["foreignProjectRejected"] = true;
                _evidence["foreignProjectRejectionStage"] = denied.Stage;
            }

            _loaderA = new HybridCLRLoader(_startup.Resources, compatibility);
            var firstA = _loaderA.LoadAsync(new ResourceAddress("hclr-sub-a", HybridCLRManifest.ManifestPath));
            yield return firstA;
            Require(firstA.Exception == null && firstA.Result != null && firstA.Result.Mode == HybridCLRLoadMode.Native && !firstA.Result.IsSimulation,
                "Conflict phase must first perform A's real native load.");
            _loaderB = new HybridCLRLoader(_startup.Resources, compatibility);
            var deniedB = _loaderB.LoadAsync(new ResourceAddress("hclr-sub-b", HybridCLRManifest.ManifestPath));
            yield return deniedB;
            Require(deniedB.Exception != null && deniedB.Result == null && deniedB.RequiresRestart && deniedB.Stage == "NativePreflight" &&
                deniedB.Exception.Message == "A different or incomplete assembly is already registered: " + SupportName,
                "A valid different-MVID shared Support must be rejected by actual native preflight before irreversible loading.");
            _evidence["sharedConflictRejected"] = true;
            _evidence["sharedConflictStage"] = deniedB.Stage;
            _evidence["sharedConflictRequiresRestart"] = deniedB.RequiresRestart;
            var cleanup = new List<Exception>();
            yield return Close(cleanup);
            Require(cleanup.Count == 0 && _startup == null && ResourceSystem.Current == null, "Conflict resource phase did not fully stop.");
            _evidence["conflictPhaseStopped"] = true;
            _evidence["barrierAReleased"] = false;
            _evidence["barrierBReleased"] = false;
            _evidence["stopped"] = false;
            var store = new ArtifactStore(Path.Combine(cacheDirectory, "positive"), new[]
            {
                new ResourceProjectMount("hclr-sub-a", "StandaloneWindows64", compatibility, snapshotA),
                new ResourceProjectMount("hclr-sub-b", "StandaloneWindows64", compatibility, snapshotB)
            });
            yield return StartStore(store);

            _loaderA = new HybridCLRLoader(_startup.Resources, compatibility);
            var requestA = _loaderA.LoadAsync(new ResourceAddress("hclr-sub-a", HybridCLRManifest.ManifestPath));
            yield return requestA;
            _loadedA = requestA.Result;
            Record("A", requestA);
            if (requestA.Exception != null)
            {
                throw new InvalidOperationException("Project A native loading failed at " + requestA.Stage, requestA.Exception);
            }

            Require(_loadedA != null && _loadedA.Mode == HybridCLRLoadMode.Native && !_loadedA.IsSimulation &&
                _loadedA.ProjectId == "hclr-sub-a" && _loadedA.CodePackageId == releaseA.Value<string>("CodePackageId") &&
                _loadedA.Assemblies.Keys.SequenceEqual(new[] { SupportName }), "A must natively load exactly Support.");

            _loaderB = new HybridCLRLoader(_startup.Resources, compatibility);
            var requestB = _loaderB.LoadAsync(new ResourceAddress("hclr-sub-b", HybridCLRManifest.ManifestPath));
            yield return requestB;
            _loadedB = requestB.Result;
            Record("B", requestB);
            if (requestB.Exception != null)
            {
                throw new InvalidOperationException("Project B native loading failed at " + requestB.Stage, requestB.Exception);
            }

            Require(_loadedB != null && _loadedB.Mode == HybridCLRLoadMode.Native && !_loadedB.IsSimulation &&
                _loadedB.ProjectId == "hclr-sub-b" && _loadedB.CodePackageId == releaseB.Value<string>("CodePackageId") &&
                _loadedB.Assemblies.Keys.OrderBy(n => n, StringComparer.Ordinal).SequenceEqual(new[] { EntryName, SupportName }),
                "B must natively load the Entry + Support dependency closure.");
            _evidence["positiveAfterConflict"] = true;
            var shared = ReferenceEquals(_loadedA.Assemblies[SupportName], _loadedB.Assemblies[SupportName]);
            _evidence["sharedSupportAssembly"] = shared;
            var supportCount = AppDomain.CurrentDomain.GetAssemblies().Count(a => a.GetName().Name == SupportName);
            _evidence["supportAssemblyCount"] = supportCount;
            Require(shared && supportCount == 1, "The same Support must be reused across both native loaders without a duplicate assembly.");

            var versionInfo = _loadedA.Assemblies[SupportName].GetType("YLFramework.Samples.HybridCLR.HotUpdate.VersionInfo", true, false);
            var values = new List<int> { 3, 5 };
            var generic = versionInfo.GetMethods(BindingFlags.Public | BindingFlags.Static).Single(m => m.Name == "Count" && m.IsGenericMethodDefinition).MakeGenericMethod(typeof(int));
            var count = (int)generic.Invoke(null, new object[] { values });
            Require(count == values.Count && count == 2, "The actual shared hot generic method did not consume its AOT List<int>.");
            _evidence["genericCount"] = count;
            var version = (string)versionInfo.GetProperty("Version", BindingFlags.Public | BindingFlags.Static).GetValue(null);
            var method = versionInfo.GetMethod("DescribeAsync", BindingFlags.Public | BindingFlags.Static, null, Type.EmptyTypes, null);
            Require(method != null && method.ReturnType == typeof(Task<string>), "Missing actual Support DescribeAsync contract.");
            var task = method.Invoke(null, null) as Task<string>;
            Require(task != null && !string.IsNullOrEmpty(version), "Actual Support did not return its version and Task<string>.");
            var deadline = Time.realtimeSinceStartup + 60;
            while (!task.IsCompleted)
            {
                Require(Time.realtimeSinceStartup < deadline, "Actual hot async continuation timed out.");
                yield return null;
            }
            var output = task.GetAwaiter().GetResult();
            _evidence["supportVersion"] = version;
            _evidence["asyncOutput"] = output;
            Require(output == version + "-async", "Actual hot async output did not match the loaded Support version.");
            var entry = _loadedB.Assemblies[EntryName];
            var moduleType = entry.GetType("YLFramework.Samples.HybridCLR.HotUpdate.ExampleModule", true, false);
            Require(moduleType.Assembly == entry && typeof(YLFramework.Core.Module).IsAssignableFrom(moduleType) &&
                !moduleType.IsAbstract && !moduleType.ContainsGenericParameters, "Entry must expose the concrete sample Module type.");
            var started = YLFramework.Core.ModuleManager.GetModule(moduleType) != null;
            _evidence["entryModuleType"] = moduleType.FullName;
            _evidence["entryModuleStarted"] = started;
            Require(!started, "Do not start the old sample Module against these new code-only data projects.");
        }

        private IEnumerator StartStore(ArtifactStore store)
        {
            _startup = new ResourceStartup(new ResourceConfiguration(store), bindCore: false);
            using var init = _startup.StartAsync();
            yield return init;
            init.TakeResult();
            Require(_startup.OwnsResourceSystem && _startup.CoreProvider == null, "The probe must own an unbound Resource startup.");
        }

        private void Record(string project, HybridCLRLoadRequest request)
        {
            ((JArray)_evidence["loads"]).Add(new JObject
            {
                ["subset"] = project,
                ["projectId"] = request.ProjectId,
                ["stage"] = request.Stage,
                ["requiresRestart"] = request.RequiresRestart,
                ["error"] = request.Exception?.ToString(),
                ["mode"] = request.Result == null ? JValue.CreateNull() : new JValue(request.Result.Mode.ToString()),
                ["simulation"] = request.Result == null ? JValue.CreateNull() : new JValue(request.Result.IsSimulation),
                ["codePackageId"] = request.Result?.CodePackageId,
                ["assemblies"] = new JArray(request.Result?.Assemblies.Keys.OrderBy(n => n, StringComparer.Ordinal) ?? Enumerable.Empty<string>())
            });
            var results = new[] { _loadedA, _loadedB }.Where(result => result != null).ToArray();
            if (results.Length != 0)
            {
                _evidence["simulation"] = results.Any(result => result.IsSimulation);
            }
        }

        private IEnumerator Close(List<Exception> failures)
        {
            try
            {
                if (_loaderB != null)
                {
                    _loaderB.Dispose();
                    _loaderB = null;
                    _evidence["barrierBReleased"] = true;
                }
            }
            catch (Exception error)
            {
                failures.Add(error);
                Debug.LogException(error);
            }
            try
            {
                if (_loaderA != null)
                {
                    _loaderA.Dispose();
                    _loaderA = null;
                    _evidence["barrierAReleased"] = true;
                }
            }
            catch (Exception error)
            {
                failures.Add(error);
                Debug.LogException(error);
            }
            if (_startup == null)
            {
                yield break;
            }

            using var stop = _startup.StopAsync();
            yield return stop;
            Require(stop.TakeResult() && ResourceSystem.Current == null, "The owned Resource startup did not stop after both barriers were released.");
            _startup = null;
            _evidence["stopped"] = true;
        }

        private static void Require(bool condition, string error)
        {
            if (!condition)
            {
                throw new InvalidOperationException(error);
            }
        }
    }
}
