using System;
using System.IO;
using System.Linq;
using System.Reflection;
using NUnit.Framework;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine;
using UnityEngine.SceneManagement;

namespace YLFramework.Tests.Integrations.HybridCLR
{
    // 只由显式 Owned 安装配置复制到独立接收端，不进入普通 Source/GUI 的测试集合。
    public class HybridCLROwnedRunnerSceneTests
    {
        [SetUp]
        public void RequireOwnedSavedHost()
        {
            Assert.That(HybridCLRTestFixtures.ShouldRegisterRunnerSceneSetup(Application.isBatchMode, Environment.GetCommandLineArgs()), Is.True);
            var setup = EditorSceneManager.GetSceneManagerSetup();
            Assert.That(setup.Length, Is.EqualTo(1));
            Assert.That(setup[0].path, Does.StartWith("Assets/HybridCLRTestsBootstrap-"));
            Assert.That(setup[0].isLoaded && setup[0].isActive, Is.True);
            Assert.That(SceneManager.GetActiveScene().isDirty, Is.False);
            Assert.That(SceneManager.GetActiveScene().rootCount, Is.Zero);
        }

        [Test]
        public void OwnedRunnerHasASavedEmptyHostBeforeCrossPlayFixtures()
        {
            var setup = EditorSceneManager.GetSceneManagerSetup();
            var scene = SceneManager.GetActiveScene();
            Assert.That(scene.path, Is.EqualTo(setup[0].path));
            Assert.That(File.Exists(scene.path), Is.True);
        }

        [TestCase("dirty")]
        [TestCase("unknown-saved")]
        [TestCase("unknown-root")]
        [TestCase("extra-component")]
        [TestCase("extra-empty")]
        [TestCase("extra-default")]
        public void SceneGuardRejectsForeignStateWithoutDiscardingIt(string state)
        {
            var host = SceneManager.GetActiveScene();
            var scene = EditorSceneManager.NewScene(state is "extra-component" or "extra-default" ? NewSceneSetup.DefaultGameObjects : NewSceneSetup.EmptyScene, NewSceneMode.Additive);
            var path = "Assets/HybridCLRRunnerGuard-" + Guid.NewGuid().ToString("N") + ".unity";
            try
            {
                if (state == "dirty")
                {
                    EditorSceneManager.MarkSceneDirty(scene);
                }
                else if (state == "unknown-saved")
                {
                    Assert.That(EditorSceneManager.SaveScene(scene, path), Is.True);
                }
                else
                {
                    if (state == "extra-component")
                    {
                        scene.GetRootGameObjects().Single(root => root.GetComponent<Camera>()).AddComponent<BoxCollider>();
                    }
                    else if (state == "unknown-root")
                    {
                        SceneManager.MoveGameObjectToScene(new GameObject("NotAnOfficialRunnerObject"), scene);
                    }
                    // 只清理自建场景标记，以覆盖官方 clean-untitled bootstrap 的布局边界。
                    var clear = typeof(EditorSceneManager).GetMethod("ClearSceneDirtiness", BindingFlags.Static | BindingFlags.NonPublic);
                    Assert.That(clear, Is.Not.Null);
                    clear.Invoke(null, new object[] { scene });
                }
                var count = SceneManager.sceneCount;
                var roots = scene.GetRootGameObjects().Select(root => root.GetInstanceID()).ToArray();
                var dirty = scene.isDirty;
                var bytes = File.Exists(path) ? File.ReadAllBytes(path) : null;
                Assert.Throws<InvalidOperationException>(() => HybridCLRTestFixtures.ValidateRunnerScenes(host.path));
                Assert.That(scene.IsValid() && scene.isLoaded, Is.True);
                Assert.That(SceneManager.sceneCount, Is.EqualTo(count));
                Assert.That(scene.isDirty, Is.EqualTo(dirty));
                CollectionAssert.AreEqual(roots, scene.GetRootGameObjects().Select(root => root.GetInstanceID()).ToArray());
                if (bytes != null)
                {
                    CollectionAssert.AreEqual(bytes, File.ReadAllBytes(path));
                }
            }
            finally
            {
                Assert.That(EditorSceneManager.CloseScene(scene, true), Is.True);
                SceneManager.SetActiveScene(host);
                if (File.Exists(path))
                {
                    AssetDatabase.DeleteAsset(path);
                }
            }
        }

        [Test]
        public void KnownDefaultRunnerObjectsAreAcceptedWithoutSavingOrDeletingThem()
        {
            var host = SceneManager.GetActiveScene();
            var scene = EditorSceneManager.NewScene(NewSceneSetup.DefaultGameObjects, NewSceneMode.Additive);
            try
            {
                Assert.That(scene.isDirty, Is.False);
                var roots = scene.GetRootGameObjects().Select(root => root.GetInstanceID()).ToArray();
                Assert.DoesNotThrow(() => HybridCLRTestFixtures.ValidateRunnerScene(scene));
                Assert.That(scene.path, Is.Empty);
                Assert.That(scene.isDirty, Is.False);
                CollectionAssert.AreEqual(roots, scene.GetRootGameObjects().Select(root => root.GetInstanceID()).ToArray());
            }
            finally
            {
                Assert.That(EditorSceneManager.CloseScene(scene, true), Is.True);
                SceneManager.SetActiveScene(host);
            }
        }
    }
}
