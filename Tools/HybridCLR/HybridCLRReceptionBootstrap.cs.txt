        private static string Argument(string name)
        {
            var args = Environment.GetCommandLineArgs();
            var at = Array.IndexOf(args, name);
            if (at < 0 || at + 1 == args.Length || at != Array.LastIndexOf(args, name))
            {
                throw new ArgumentException("One explicit " + name + " is required.");
            }

            return args[at + 1];
        }
        private static JObject Owned()
        {
            RequireSubset(IsExplicitReceiverPath(SourceRepository) && Directory.Exists(SourceRepository), "Install or sync this consumer from its trusted source repository before use.");
            RejectSubsetLinks(SourceRepository);
            RequireSubset(SameSubsetPath(Root, ExpectedRoot), "Wrong installed consumer evidence root.");
            RejectSubsetLinks(Root + "/owner.json");
            var owner = JObject.Parse(File.ReadAllText(Root + "/owner.json"));
            var compatibility = BoundOwnerSha256.Length != 0;
            RequireSubset(!compatibility || Sha(Root + "/owner.json") == BoundOwnerSha256, "The exact installed owner binding changed.");
            var baseImport = compatibility ? owner.Value<string>("baseImportRun") : "base-no-package-import";
            var optionalImport = compatibility ? owner.Value<string>("optionalImportRun") : "optional-no-package-import";
            foreach (var name in new[] { baseImport, optionalImport })
            {
                RejectSubsetLinks(Root + "/" + name + ".receipt.json");
                var receipt = JObject.Parse(File.ReadAllText(Root + "/" + name + ".receipt.json"));
                RequireSubset(receipt.Value<bool>("passed") && !receipt.Value<bool>("testsPresent"), "Product-only imports are required before consumers.");
                RequireBoundReceipt(owner, receipt);
            }
            RequireSubset(IsExplicitReceiverPath(owner.Value<string>("repository")) && SameSubsetPath(owner.Value<string>("repository"), SourceRepository), "Receiver source repository does not match the installed consumer binding.");
            if (!Application.isBatchMode || Application.unityVersion != ExpectedUnityVersion ||
                owner.Value<string>("purpose") != "YLFramework-HybridCLR-9.0.3-independent-reception" ||
                !IsExplicitReceiverPath(owner.Value<string>("project")) || !SameSubsetPath(owner.Value<string>("project"), Project) || Directory.Exists(Project + "/.git"))
            {
                throw new InvalidOperationException("Never run this acceptance consumer on the source project or before product-only imports.");
            }

            if (compatibility)
            {
                RequireSubset(owner.Value<int>("schemaVersion") == 2 && !owner.Value<bool>("simulation") &&
                    SameSubsetPath(owner.Value<string>("deliveryRoot"), "G:/Unity/U63-20261027-c63a1") &&
                    SameSubsetPath(owner.Value<string>("evidenceRoot"), Root) &&
                    SameSubsetPath(Project, Path.Combine(owner.Value<string>("deliveryRoot"), owner.Value<string>("receiver"))) &&
                    SameSubsetPath(EditorApplication.applicationPath, owner.Value<string>("unity")) &&
                    System.Diagnostics.FileVersionInfo.GetVersionInfo(EditorApplication.applicationPath).ProductVersion == ExpectedUnityVersion + "_" + ExpectedUnityRevision,
                    "The live Editor/revision must match the immutable owned receiver, not a simulated owner.");
                foreach (var name in new[] { "project", "baseline", "releaseBaseline", "playerRoot", "releasePlayerRoot", "runtimeCacheRoot", "subsetBaseline", "subsetPlayerRoot", "subsetCacheRoot" })
                {
                    RejectSubsetLinks(owner.Value<string>(name));
                }
                RejectSubsetLinks(Project);
                return owner;
            }
            var durable = SameSubsetPath(Root, "G:/Unity/YLStyleAcceptance/YS28");
            var approved = Path.GetFullPath(Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData), "Temp/kilo")) + Path.DirectorySeparatorChar;
            RequireSubset(Root.StartsWith(approved, StringComparison.OrdinalIgnoreCase) || durable, "Expected an explicitly approved independent receiver root.");
            var parent = Directory.GetParent(SourceRepository).Parent.FullName;
            var structure = owner.Value<string>("work") == "yl-structure-and-namespaces-20261023";
            if (owner.Property("nativePrefix") != null || structure || owner.Value<string>("work") == "yl-code-style-naming-cleanup-20261022" || durable)
            {
                var prefix = owner.Value<string>("nativePrefix");
                RequireSubset(!string.IsNullOrEmpty(prefix) && System.Text.RegularExpressions.Regex.IsMatch(prefix, structure ? "^S[0-9]{2}$" : "^YS[0-9]{2}$") &&
                    Path.GetFileName(Root.TrimEnd(Path.DirectorySeparatorChar, Path.AltDirectorySeparatorChar)) == prefix &&
                    (structure || owner.Value<string>("work") == "yl-code-style-naming-cleanup-20261022") &&
                    !SameSubsetPath(SourceRepository, Project), "An exact isolated receiver owner is required; never fall back to historical outputs.");
                RequireSubset(SameSubsetPath(Project, Path.Combine(Root, "R")) || SameSubsetPath(Project, Path.Combine(parent, prefix)), "Unexpected isolated receiver project.");
                var paths = new Dictionary<string, string>
                {
                    ["playerRoot"] = Path.Combine(parent, prefix + "D4"),
                    ["releasePlayerRoot"] = Path.Combine(parent, prefix + "R"),
                    ["runtimeCacheRoot"] = Path.Combine(parent, prefix + "C"),
                    ["subsetPlayerRoot"] = Path.Combine(parent, prefix + "S"),
                    ["subsetCacheRoot"] = Path.Combine(parent, prefix + "CS"),
                    ["baseline"] = structure ? Path.Combine(parent, prefix + "B") : Path.Combine(Root, "Baselines/Native"),
                    ["subsetBaseline"] = structure ? Path.Combine(parent, prefix + "BS") : Path.Combine(Root, "Baselines/ProjectSubsets")
                };
                foreach (var path in paths)
                {
                    var value = owner.Value<string>(path.Key);
                    RequireSubset(IsExplicitReceiverPath(value) && SameSubsetPath(value, path.Value), "Unowned isolated receiver path: " + path.Key);

        private static string ScopedOwnerValue(string name, string legacyValue)
        {
            var owner = Owned();
            return BoundOwnerSha256.Length == 0 && owner.Property("nativePrefix") == null ? legacyValue : owner.Value<string>(name);
        }
        private static void RequireBoundReceipt(JObject owner, JObject receipt)
        {
            if (BoundOwnerSha256.Length != 0)
            {
                RequireSubset(receipt.Value<string>("UnityVersion") == ExpectedUnityVersion && receipt.Value<string>("UnityRevision") == ExpectedUnityRevision &&
                    receipt.Value<string>("Receiver") == owner.Value<string>("receiver") && receipt.Value<string>("OwnerSha256") == BoundOwnerSha256,
                    "Cross-engine or foreign-owner baseline/receipt rejected.");
            }
        }
        private static string Sha(string path)
        {
            using var sha = SHA256.Create();
            using var stream = File.OpenRead(path);
            return BitConverter.ToString(sha.ComputeHash(stream)).Replace("-", "").ToLowerInvariant();
        }
        private static void Receipt(string name, object value)
        {
            RequireSubset(!string.IsNullOrEmpty(name) && System.Text.RegularExpressions.Regex.IsMatch(name, "^[A-Za-z0-9-]+$"), "Use one receipt name, not a path.");
            var path = Root + "/" + name + ".json";
            RejectSubsetLinks(path);
            if (File.Exists(path))
            {
                throw new IOException("Preserve the prior acceptance receipt.");
            }

            var pending = path + ".pending-" + Guid.NewGuid().ToString("N");
            RejectSubsetLinks(pending);
            using (var stream = new FileStream(pending, FileMode.CreateNew, FileAccess.Write, FileShare.None))
            {
                var bytes = Encoding.UTF8.GetBytes(JsonConvert.SerializeObject(value, Formatting.Indented));
                stream.Write(bytes, 0, bytes.Length);
                stream.Flush(true);
            }
            File.Move(pending, path);
        }
        private static string ProductPackageSha(JObject owner, bool optional)
        {
            if (BoundOwnerSha256.Length == 0)
            {
                return optional ? JObject.Parse(File.ReadAllText(Root + "/product-selection.json")).Value<string>("packageSha256") : Sha(Root + "/YLFramework.unitypackage");
            }
            var path = Path.Combine(owner.Value<string>("deliveryRoot"), owner.Value<string>(optional ? "optionalPackage" : "basePackage"));
            var receiptPath = Root + "/" + owner.Value<string>(optional ? "optionalImportRun" : "baseImportRun") + ".receipt.json";
            RejectSubsetLinks(path);
            RejectSubsetLinks(receiptPath);
            var proof = JObject.Parse(File.ReadAllText(receiptPath));
            RequireBoundReceipt(owner, proof);
            var sha = Sha(path);
            RequireSubset(proof.Value<bool>("passed") && !proof.Value<bool>("testsPresent") && proof.Value<string>("action") == (optional ? "ImportOptional" : "ImportBase") &&
                string.Equals(proof.Value<string>("packageSha256"), sha, StringComparison.OrdinalIgnoreCase),
                "The formal frozen package must still match this receiver's product-only import proof.");
            return sha;
        }

        public static void Install()
        {
            var owner = Owned();
            RequireSubset(owner["nativeSourceManifest"] == null, "A maintained-native owner must use InstallLocal, never the default downloader.");
            var package = UnityEditor.PackageManager.PackageInfo.GetAllRegisteredPackages().Single(p => p.name == "com.code-philosophy.hybridclr");
            var locked = JObject.Parse(File.ReadAllText("Packages/packages-lock.json"))["dependencies"]["com.code-philosophy.hybridclr"];
            if (package.version != "9.0.3" || package.source != UnityEditor.PackageManager.PackageSource.Git ||
                locked.Value<string>("version") != "https://gitee.com/focus-creative-games/hybridclr_unity.git#v9.0.3" ||
                locked.Value<string>("hash") != "6cadd24d101f4cbd00cd80190106f6621132d24d")
            {
                throw new InvalidDataException("Wrong official release.");
            }

            new global::HybridCLR.Installer.InstallerController().InstallDefaultHybridCLR();
            var installed = new global::HybridCLR.Installer.InstallerController();
            if (!installed.HasInstalledHybridCLR() || installed.PackageVersion != "9.0.3" || installed.InstalledLibil2cppVersion != "9.0.3")
            {
                throw new InvalidOperationException("Explicit official runtime installation failed.");
            }

            Receipt(Run, new
            {
                passed = true,
                package.version,
                hash = locked.Value<string>("hash"),
                installed.PackageVersion,
                installed.InstalledLibil2cppVersion
            });
        }
        private static JObject NativeSource(JObject owner, bool installed)
        {
            if (owner["nativeSourceManifest"] == null)
            {
                return null;
            }
            var path = owner.Value<string>("nativeSourceManifest");
            RejectSubsetLinks(path);
            RequireSubset(ExpectedUnityVersion == "6000.3.27f1" && Sha(path) == owner.Value<string>("nativeSourceManifestSha256") &&
                Path.GetFullPath(path).StartsWith(Path.GetFullPath(Path.Combine(owner.Value<string>("deliveryRoot"), "B08Native")) + Path.DirectorySeparatorChar, StringComparison.OrdinalIgnoreCase),
                "Foreign or changed maintained-native source receipt.");
            var source = JObject.Parse(File.ReadAllText(path));
            var patched = source.Value<string>("Variant") == "ParameterOwner";
            RequireSubset(source.Value<string>("Schema") == "B08NativeSource-v1" && (patched || source.Value<string>("Variant") == "UnmodifiedControl") &&
                source.Value<string>("SdkCommit") == "6cadd24d101f4cbd00cd80190106f6621132d24d" &&
                source.Value<string>("Il2CppCommit") == "6eb38b29c3a7448b3ac6c4dee450de375b5d158d" &&
                source.Value<string>("HybridClrCommit") == "bf01f8eed56a077781c3a7b25f418603c7645588" &&
                source["ChangedFiles"].Values<string>().SequenceEqual(patched ? new[] { "vm/GlobalMetadata.cpp" } : Array.Empty<string>()), "Unapproved native variant.");
            var directory = installed ? global::HybridCLR.SettingsUtil.LocalIl2CppDir + "/libil2cpp" : source.Value<string>("SourceDirectory");
            RejectSubsetLinks(directory);
            foreach (var file in source["Files"].OfType<JObject>())
            {
                var relative = file.Value<string>("Path");
                RequireSubset(!Path.IsPathRooted(relative) && !relative.Contains("..") && !relative.Contains('\\') && !relative.Contains(':'), "Escaping native source file.");
                if (installed && relative.StartsWith("hybridclr/generated/", StringComparison.Ordinal))
                {
                    continue; // The official Prepare stages own generated bridges; source implementation bytes must remain exact.
                }
                var actual = Path.Combine(directory, relative);
                RejectSubsetLinks(actual);
                RequireSubset(new FileInfo(actual).Length == file.Value<long>("Length") && Sha(actual) == file.Value<string>("Sha256"), "Maintained native source drift: " + relative);
            }
            return source;
        }

                locked.Value<string>("hash") == source.Value<string>("SdkCommit") && !Directory.Exists(global::HybridCLR.SettingsUtil.LocalUnityDataDir) &&
                !global::HybridCLR.Settings.HybridCLRSettings.instance.useGlobalIl2cpp, "A new receiver and exact fixed official SDK local installer are required.");
            installer.InstallFromLocal(source.Value<string>("SourceDirectory"));
            NativeSource(owner, true);
            Receipt("native-source-binding", new { passed = true, Variant = source.Value<string>("Variant"), SourceManifestSha256 = owner.Value<string>("nativeSourceManifestSha256"),
                OwnerSha256 = BoundOwnerSha256, Receiver = owner.Value<string>("receiver"), UnityVersion = Application.unityVersion, UnityRevision = ExpectedUnityRevision,

            Receipt(Run, new { passed = true, Variant = source.Value<string>("Variant"), SourceManifestSha256 = owner.Value<string>("nativeSourceManifestSha256"), OfficialLocalInstaller = true });
        }

        public static void CompileExternalNewRequirement()
        {
            Owned();
            var path = Root + "/ExternalSource/VersionInfo.cs";
            var backup = path + ".v2-owned-backup";
            if (File.Exists(backup))
            {
                throw new IOException("Preserve the existing v2 source backup.");
            }

            var text = File.ReadAllText(path);
            if (!text.Contains("\"v2\""))
            {
                throw new InvalidDataException("Run the verified v2 patch before the new-native-requirement rejection fixture.");
            }

            File.Copy(path, backup, false);
            text = text.Replace("public static string Version", "[System.Runtime.InteropServices.DllImport(\"hclr-new-native-requirement-fixture\")] public static extern void NewNativeRequirement();\n        public static string Version");
            File.WriteAllText(path, text, new UTF8Encoding(false));
            CompileSupport(false);
        }

        }
        private static void CompileSupport(bool patch, bool release = false)
        {
            Owned();
            if (BoundOwnerSha256.Length != 0 && (EditorUserBuildSettings.activeBuildTarget != BuildTarget.StandaloneWindows64 ||
                PlayerSettings.GetScriptingBackend(BuildTargetGroup.Standalone) != ScriptingImplementation.IL2CPP ||
                PlayerSettings.GetApiCompatibilityLevel(BuildTargetGroup.Standalone) != ApiCompatibilityLevel.NET_4_6))
            {
                // Unity 6.3 rebuilds AssemblyBuilder references from the actual PlayerSettings profile.
                throw new InvalidOperationException("Configure the actual Windows64 IL2CPP/.NET Framework receiver before external compilation; compilerOptions do not override the active API profile.");
            }
            var source = Root + "/ExternalSource/VersionInfo.cs";
            if (!File.Exists(source))
            {
                Directory.CreateDirectory(Path.GetDirectoryName(source));
                File.Copy(SampleRoot + "/HotSupport/VersionInfo.cs", source, false);
            }
            if (patch)
            {
                var text = File.ReadAllText(source);
                if (!text.Contains("\"v1\""))
                {
                    throw new InvalidDataException("Expected the owned v1 external source.");
                }

                File.WriteAllText(source, text.Replace("\"v1\"", "\"v2\""), new UTF8Encoding(false));
            }
            var directory = Root + "/ExternalHotUpdate/StandaloneWindows64";
            Directory.CreateDirectory(directory);
            var output = directory + "/" + SupportName + ".dll";
            var builder = new AssemblyBuilder(output, new[] { source })
            {
                buildTarget = BuildTarget.StandaloneWindows64,
                buildTargetGroup = BuildTargetGroup.Standalone,
                flags = release ? AssemblyBuilderFlags.None : AssemblyBuilderFlags.DevelopmentBuild,
                compilerOptions = new ScriptCompilerOptions { ApiCompatibilityLevel = ApiCompatibilityLevel.NET_4_6 }
            };
            builder.buildFinished += (path, messages) =>
            {
                try
                {
                    var errors = messages.Where(m => m.type == CompilerMessageType.Error).Select(m => m.message).ToArray();
                    Receipt(Run, new
                    {
                        passed = errors.Length == 0 && File.Exists(path),
                        sourceSha256 = Sha(source),
                        dllSha256 = File.Exists(path) ? Sha(path) : null,
                        errors
                    });
                    if (errors.Length != 0 || !File.Exists(path))
                    {
                        throw new InvalidOperationException(string.Join("\n", errors));
                    }

                    EditorApplication.delayCall += () => EditorApplication.Exit(0);
                }
                catch (Exception error)
                {
                    Debug.LogException(error);
                    EditorApplication.delayCall += () => EditorApplication.Exit(1);
                }
            };
            if (!builder.Build())
            {
                throw new InvalidOperationException("The explicit external DLL compiler could not start.");
            }
        }

        public static void Setup()
        {
            var owner = Owned();
            if (Directory.Exists(AssetsRoot))
            {
                throw new IOException("Never replace an existing sample setup.");
            }

            if (PlayerSettings.GetScriptingBackend(BuildTargetGroup.Standalone) != ScriptingImplementation.IL2CPP || !EditorUserBuildSettings.development)
            {
                throw new InvalidOperationException("Run Configure as a separate invocation before authoring settled assets.");
            }

            var profile = HybridCLRExampleAssets.Create(AssetsRoot, owner.Value<string>("baseline"));
            if (BoundOwnerSha256.Length != 0)
            {
                profile.recipe.appContentCompatibilityId = owner.Value<string>("compatibilityId");
                EditorUtility.SetDirty(profile.recipe);
                AssetDatabase.SaveAssetIfDirty(profile.recipe);
            }
            var scene = EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);
            var runner = new GameObject("HybridCLR Native Acceptance").AddComponent<HybridCLRReceptionRunner>();
            runner.projectId = profile.recipe.project.projectId;
            runner.compatibilityId = profile.recipe.appContentCompatibilityId;
            if (!EditorSceneManager.SaveScene(scene, AssetsRoot + "/Acceptance.unity"))
            {
                throw new IOException("Acceptance scene was not saved.");
            }

            EditorBuildSettings.scenes = new[] { new EditorBuildSettingsScene(AssetsRoot + "/Acceptance.unity", true) };
            AssetDatabase.StartAssetEditing();
            try
            {
                var entryPath = SampleRoot + "/HotEntry/" + EntryName + ".asmdef";
                var entry = JObject.Parse(File.ReadAllText(entryPath));
                entry["references"] = new JArray(entry["references"].Values<string>().Where(n => n != SupportName));
                entry["precompiledReferences"] = new JArray(SupportName + ".dll");
                File.WriteAllText(entryPath, entry.ToString(Formatting.Indented), new UTF8Encoding(false));
                var archive = Root + "/SourceSupportArchive";
                if (Directory.Exists(archive))
                {
                    throw new IOException("Preserve the original sample source archive.");
                }

                Directory.Move(SampleRoot + "/HotSupport", archive);
                File.Move(SampleRoot + "/HotSupport.meta", archive + ".meta");
                Directory.CreateDirectory("Assets/HCLRExternal");
                File.Copy(Root + "/ExternalHotUpdate/StandaloneWindows64/" + SupportName + ".dll", "Assets/HCLRExternal/" + SupportName + ".dll", false);
            }
            finally
            {
                AssetDatabase.StopAssetEditing();
            }
            AssetDatabase.Refresh(ImportAssetOptions.ForceSynchronousImport);
            Receipt(Run, new

        public static void RecoverPending()
        {
            Owned();
            var profile = Profile();
            var root = Path.GetFullPath(profile.generatedAssetDirectory);
            var owner = JObject.Parse(File.ReadAllText(root + "/hybridclr-owner.json"));
            if (owner.Value<string>("ProfileGuid") != AssetDatabase.AssetPathToGUID(AssetDatabase.GetAssetPath(profile)))
            {
                throw new InvalidDataException("Wrong generated owner.");
            }

            var files = owner["Files"].ToDictionary(f => f.Value<string>("Path"), StringComparer.Ordinal);
            foreach (var file in files)
            {
                if (Sha(root + "/" + file.Key) != file.Value.Value<string>("Sha256"))
                {
                    throw new InvalidDataException("A prior owned payload changed; recovery refused: " + file.Key);
                }
            }

            var removed = new List<object>();
            foreach (var path in Directory.GetFiles(root, "*.pending-*", SearchOption.AllDirectories))
            {
                var match = System.Text.RegularExpressions.Regex.Match(path, @"^(.*)\.pending-[a-f0-9]{32}$");
                if (!match.Success)
                {
                    throw new InvalidDataException("Unknown pending path.");
                }

                var target = match.Groups[1].Value;
                var relative = target.Substring(root.Length + 1).Replace('\\', '/');
                if (!files.ContainsKey(relative))
                {
                    throw new InvalidDataException("Pending target is outside the ownership record.");
                }

                var digest = Sha(path);
                var assemblyName = Path.GetFileName(target).Replace(".dll.bytes", "");
                var candidates = Directory.GetFiles(Project + "/Temp/YLHybridCLR", assemblyName + ".dll", SearchOption.AllDirectories)
                    .Concat(Directory.GetFiles(Root + "/ExternalHotUpdate", assemblyName + ".dll", SearchOption.AllDirectories));
                var aot = Project + "/Library/HybridCLR/AssembliesPostIl2CppStrip/StandaloneWindows64/" + assemblyName + ".dll";
                if (File.Exists(aot))
                {
                    candidates = candidates.Append(aot);
                }

                if (!candidates.Any(p => Sha(p) == digest))
                {
                    throw new InvalidDataException("Pending bytes do not match an explicit prior compilation input.");
                }

                var asset = profile.generatedAssetDirectory + "/" + path.Substring(root.Length + 1).Replace('\\', '/');
                removed.Add(new
                {
                    path = relative,
                    pendingSha256 = digest,
                    priorSha256 = Sha(target),
                    attributes = File.GetAttributes(target).ToString()
                });
                if (!AssetDatabase.DeleteAsset(asset))
                {
                    throw new IOException("Could not remove the verified self-owned pending artifact.");
                }
            }
            Receipt(Run, new
            {
                passed = true,
                removed
            });
        }

        public static void DiagnoseResource()
        {
            Owned();
            var profile = Profile();
            var projectGuid = AssetDatabase.AssetPathToGUID(AssetDatabase.GetAssetPath(profile.recipe.project));
            var projectId = profile.recipe.project.projectId;
            Dictionary<string, string> before = null;
            Dictionary<string, string> after = null;
            Dictionary<string, string> Capture()
            {
                var values = new Dictionary<string, string>(StringComparer.Ordinal);
                var assemblies = CompilationPipeline.GetAssemblies();
                foreach (var assembly in assemblies)
                {
                    values["assembly:" + assembly.name] = assembly.flags + "|" + string.Join(";", assembly.defines.OrderBy(s => s, StringComparer.Ordinal));
                }

                var paths = Directory.GetFiles(Project + "/ProjectSettings", "*", SearchOption.AllDirectories)
                    .Concat(new[] { Project + "/Packages/manifest.json", Project + "/Packages/packages-lock.json" })
                    .Concat(assemblies.SelectMany(a => a.sourceFiles)).Concat(assemblies.SelectMany(a => a.compiledAssemblyReferences))
                    .Concat(assemblies.Select(a => CompilationPipeline.GetAssemblyDefinitionFilePathFromAssemblyName(a.name)).Where(p => !string.IsNullOrEmpty(p)));
                foreach (var path in paths.Select(Path.GetFullPath).Distinct(StringComparer.OrdinalIgnoreCase))
                {
                    if (File.Exists(path))
                    {
                        values["file:" + path] = Sha(path);
                        if (File.Exists(path + ".meta"))
                        {
                            values["file:" + path + ".meta"] = Sha(path + ".meta");
                        }
                    }
                }

                var map = ResourceBuildMap.Create(AssetDatabase.LoadAssetAtPath<ResourceProject>(AssetDatabase.GUIDToAssetPath(projectGuid)));
                var fingerprintPaths = (IEnumerable<string>)typeof(ResourceBuildMap).GetProperty("FingerprintPaths", System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.NonPublic).GetValue(map);
                foreach (var path in fingerprintPaths)
                {
                    values["asset-hash:" + path] = AssetDatabase.GetAssetDependencyHash(path).ToString();
                    if (File.Exists(path))
                    {
                        values["asset-file:" + path] = Sha(path);
                    }
                }
                return values;
            }
            void Observe(ResourceBuildStage stage)
            {
                if (stage.ProjectId != projectId)
                {
                    return;
                }

                if (stage.Phase == ResourceBuildPhase.NativeBuildStarted)
                {
                    before = Capture();
                }

                if (stage.Phase == ResourceBuildPhase.ReadbackStarted)
                {
                    after = Capture();
                }
            }
            Exception failure = null;
            ResourceBuildProgress.StageReached += Observe;
            try
            {
                ResourceBundleBuilder.Build(profile.recipe.project, profile.recipe.target, true);
            }
            catch (Exception error)
            {
                failure = error;
            }
            finally
            {
                ResourceBuildProgress.StageReached -= Observe;
            }
            var changed = before == null || after == null ? null : before.Keys.Union(after.Keys).Where(k => !before.TryGetValue(k, out var prior) || !after.TryGetValue(k, out var current) || prior != current)
                .Select(k => new { key = k, before = before.TryGetValue(k, out var prior) ? prior : null, after = after.TryGetValue(k, out var current) ? current : null }).ToArray();
            Receipt(Run, new
            {
                passed = failure == null,
                error = failure?.ToString(),
                changed
            });
            if (failure != null)
            {
                throw failure;
            }
        }

        private static object ReadParameterOracle(string baseline)
        {
            var path = Path.Combine(baseline, "Aot/mscorlib.dll");
            RejectSubsetLinks(path);
            using var module = dnlib.DotNet.ModuleDefMD.Load(path);
            var method = module.Types.Single(t => t.FullName == "System.Array").Methods.Single(m => m.Name.String == "InternalArray__RemoveAt" && m.MethodSig.Params.Count == 1);
            var parameter = method.ParamDefs.Single(p => p.Sequence == 1);
            RequireSubset(parameter.Name.String == "index" && method.MethodSig.Params[0].FullName == "System.Int32", "Final stripped AOT parameter oracle differs.");
            return new { Name = parameter.Name.String, Type = method.MethodSig.Params[0].FullName, Token = parameter.MDToken.Raw,
                MethodToken = method.MDToken.Raw, AssemblySha256 = Sha(path), Origin = "This run's frozen final stripped Aot/mscorlib.dll" };
        }

            foreach (var path in new[] { SubsetCache, Root + "/project-subsets-patch.json", Project + "/AssetBundles/YLFramework/hclr-sub-b" })
            {
                RequireNewSubsetPath(path);
            }

            RequireUnusedSubsetGenerated("B");
            HybridCLRBuildReport report = null;
            Exception failure = null;
            try
            {
                RequireSubsetEnvironment();
                var native = ReadSubsetNative(owner);
                var profiles = SubsetProfiles(owner);
                var profile = profiles[1];
                var baseline = JObject.Parse(File.ReadAllText(Root + "/project-subsets-base.json"));
                RequireBoundReceipt(owner, baseline);
                if (native != null)
                {
                    RequireSubset(baseline.Value<string>("NativeVariant") == "ParameterOwner" &&
                        baseline.Value<string>("NativeSourceManifestSha256") == owner.Value<string>("nativeSourceManifestSha256"), "Do not publish a subset against a foreign native source variant.");
                }
                RequireSubset(baseline.Value<bool>("passed") && baseline.Value<bool>("Development") && baseline.Value<string>("CompatibilityId") == SubsetCompatibility &&
                    SameSubsetPath(baseline.Value<string>("PlayerDirectory"), SubsetPlayer) && SameSubsetPath(baseline.Value<string>("Baseline"), profile.baselineDirectory) &&
                    baseline.Value<string>("ProfileA") == AssetDatabase.GetAssetPath(profiles[0]) && baseline.Value<string>("ProfileB") == AssetDatabase.GetAssetPath(profile),
                    "Use this round's exact successful base receipt and explicit owner binding.");
                var beforePlayer = SubsetFingerprint(SubsetPlayer);
                var beforeBaseline = SubsetFingerprint(profile.baselineDirectory);
                var baselineSha = Sha(profile.baselineDirectory + "/baseline.json");
                RequireSubset(beforePlayer == baseline.Value<string>("PlayerFingerprint") && beforeBaseline == baseline.Value<string>("BaselineFingerprint") &&
                    baselineSha == baseline.Value<string>("BaselineSha256"), "The frozen Player or baseline changed before publication.");
                var releaseA = ResourcePublisher.ReadVersion(profiles[0].recipe.project, BuildTarget.StandaloneWindows64, SubsetCompatibility, baseline["ReleaseA"].Value<string>("CatalogDigest"));
                var publishedA = ReadSubsetRelease(profiles[0], releaseA);
                RequireSubset(JToken.DeepEquals(SubsetReleaseIdentity(releaseA, publishedA), baseline["ReleaseA"]) &&
                    publishedA.ToJson() == HybridCLRManifest.Parse(baseline["ManifestA"].ToString(Formatting.None)).ToJson(), "A's actual release or source manifest changed.");
                RequireSubsetManifest(publishedA, "hclr-sub-a", SupportName);
                ReadSubsetRegistration(profiles[0]);
                string afterPlayer = null;
                string afterBaseline = null;
                try
                {
                    report = HybridCLRBuildPipeline.PublishPatch(profile);
                }
                finally
                {
                    afterPlayer = SubsetFingerprint(SubsetPlayer);
                    afterBaseline = SubsetFingerprint(profile.baselineDirectory);
                    RequireSubset(beforePlayer == afterPlayer && beforeBaseline == afterBaseline &&
                        baselineSha == Sha(profile.baselineDirectory + "/baseline.json"), "Patch publication modified the frozen Player or any baseline file.");
                }
                RequireSubset(report.HasSucceeded, report.Summary);
                profiles = SubsetProfiles(owner);
                profile = profiles[1];
                var allowed = new[] { "Preflight", "ReadBaseline", "Compile", "PatchCompatibility", "PayloadCommit", "ResourcePublish" };
                RequireSubset(report.PlayerBuild == null && report.Deployment == null && SubsetNativeStages(report).Length == 0 && report.Stages.All(s => allowed.Contains(s.Name)),
                    "A subset patch must not build/strip a Player, regenerate native bridges/link data or deploy built-ins.");
                RequireSubsetManifest(report.Manifest, "hclr-sub-b", EntryName, SupportName);
                RequireSubset(report.Manifest.HotUpdateAssemblies.Single(p => p.Name == EntryName).Dependencies.SequenceEqual(new[] { SupportName }), "Entry's Support dependency must be in B's closure.");
                var publishedB = ReadSubsetRelease(profile, report.Release);
                RequireSubset(publishedB.ToJson() == report.Manifest.ToJson(), "B's actual published manifest differs from the patch report.");
                var support = publishedB.HotUpdateAssemblies.Single(p => p.Name == SupportName);
                RequireSameSubsetSupport(publishedA.HotUpdateAssemblies.Single(), support);
                RequireSameSubsetSupport(baseline["SharedSupport"].ToObject<HybridCLRPayload>(), support);
                RequireSubset(SameSubsetPath(report.BaselineDirectory, profile.baselineDirectory), "The patch did not use A's exact frozen baseline.");
                Receipt("project-subsets-patch", new
                {
                    passed = true,
                    UnityVersion = Application.unityVersion,
                    UnityRevision = ExpectedUnityRevision,
                    Receiver = owner.Value<string>("receiver"),
                    OwnerSha256 = BoundOwnerSha256,
                    ProfileA = AssetDatabase.GetAssetPath(profiles[0]),
                    ProfileB = AssetDatabase.GetAssetPath(profile),
                    Scene = SubsetScene,
                    PlayerDirectory = Path.GetFullPath(SubsetPlayer),
                    RuntimeCacheDirectory = Path.GetFullPath(SubsetCache),
                    Baseline = report.BaselineDirectory,
                    BaselineSha256 = baselineSha,
                    BaselineFingerprint = afterBaseline,
                    PlayerFingerprint = afterPlayer,
                    PlayerFingerprintBefore = beforePlayer,
                    BaselineFingerprintBefore = beforeBaseline,
                    CompatibilityId = SubsetCompatibility,
                    Development = profile.development,
                    ReleaseA = SubsetReleaseIdentity(releaseA, publishedA),
                    ReleaseB = SubsetReleaseIdentity(report.Release, publishedB),
                    ManifestB = JObject.Parse(publishedB.ToJson()),
                    SharedSupport = support,
                    nativeStages = SubsetNativeStages(report),
                    unchangedPlayer = beforePlayer == afterPlayer,
                    unchangedBaseline = beforeBaseline == afterBaseline,
                    NativeVariant = native?.Value<string>("Variant") ?? "OfficialSdkLocal",
                    NativeSourceManifestSha256 = owner.Value<string>("nativeSourceManifestSha256"),
                    SubsetConsumerVersion = 2
                });
            }
            catch (Exception error)
            {
                failure = error;
                throw;
            }
            finally
            {
                WriteSubsetReport(report, failure);
            }
        }
        private static object InvokeSubsetContract(MethodInfo method, params object[] arguments)
        {
            try { return method.Invoke(null, arguments); }
            catch (TargetInvocationException error) when (error.InnerException != null)
            {
                System.Runtime.ExceptionServices.ExceptionDispatchInfo.Capture(error.InnerException).Throw();
                throw;
            }
        }
        public static void VerifyProjectSubsetConstraints()
        {
            var owner = Owned();
            RequireSubsetRun();
            RequireSubsetEnvironment();
            ReadSubsetNative(owner);
            var profiles = SubsetProfiles(owner);
            var beforePlayer = SubsetFingerprint(SubsetPlayer);
            var beforeBaseline = SubsetFingerprint(SubsetBaseline(owner));
            var pipeline = typeof(HybridCLRBuildPipeline);
            var baselineType = pipeline.Assembly.GetType("YLFramework.Integrations.HybridCLR.Editor.HybridCLRBuildBaseline", true);
            var frozen = InvokeSubsetContract(baselineType.GetMethod("Read", BindingFlags.Static | BindingFlags.NonPublic), SubsetBaseline(owner));
            var descriptor = JObject.Parse(File.ReadAllText(SubsetBaseline(owner) + "/baseline.json"));
            object CloneBaseline() => descriptor.ToObject(baselineType);
            var binding = pipeline.GetMethod("ValidateBaselineBinding", BindingFlags.Static | BindingFlags.NonPublic);
            var validateSet = pipeline.GetMethod("ValidatePatchSet", BindingFlags.Static | BindingFlags.NonPublic);
            var aliases = InvokeSubsetContract(pipeline.GetMethod("UnityAotAliases", BindingFlags.Static | BindingFlags.NonPublic));
            var analyze = typeof(HybridCLRAssemblyAnalysis).GetMethods(BindingFlags.Public | BindingFlags.NonPublic | BindingFlags.Static).Single(m => m.Name == "Analyze" && m.GetParameters().Length == 3);
            var hot = descriptor["InitialManifest"]["HotUpdateAssemblies"].Select(p => SubsetBaseline(owner) + "/HotUpdate/" + p.Value<string>("Name") + ".dll").ToArray();
            var aot = descriptor["Files"].Where(f => f.Value<string>("Path").StartsWith("Aot/", StringComparison.Ordinal)).Select(f => SubsetBaseline(owner) + "/" + f.Value<string>("Path")).ToArray();
            var inputs = InvokeSubsetContract(analyze, hot, aot, aliases);
            var fingerprint = descriptor.Value<string>("SettingsFingerprint");
            InvokeSubsetContract(binding, profiles[1], frozen, fingerprint);
            InvokeSubsetContract(validateSet, frozen, inputs);
            var cases = new JArray { new JObject { ["Case"] = "valid-one-hop-binding", ["Passed"] = true }, new JObject { ["Case"] = "valid-global-metadata-native-set", ["Passed"] = true } };
            void Reject(string name, Action action)
            {
                try { action(); }
                catch (InvalidDataException error)
                {
                    cases.Add(new JObject { ["Case"] = name, ["Rejected"] = true, ["Message"] = error.Message });
                    return;
                }
                throw new InvalidOperationException("A real subset constraint unexpectedly accepted: " + name);
            }
            foreach (var name in new[] { "unbound", "self", "chain", "directory", "compatibility", "development", "target" })
            {
                var child = UnityEngine.Object.Instantiate(profiles[1]);
                var recipe = UnityEngine.Object.Instantiate(profiles[1].recipe);
                var parent = UnityEngine.Object.Instantiate(profiles[0]);
                try
                {
                    child.recipe = recipe;
                    switch (name)
                    {
                        case "unbound": child.baseProfile = null; break;
                        case "self": child.baseProfile = child; break;
                        case "chain": parent.baseProfile = profiles[0]; child.baseProfile = parent; break;
                        case "directory": child.baselineDirectory += "-foreign"; break;
                        case "compatibility": recipe.appContentCompatibilityId += "-foreign"; break;
                        case "development": child.development = !child.development; break;
                        case "target": recipe.target = BuildTarget.Android; break;
                    }
                    Reject("binding-" + name, () => InvokeSubsetContract(binding, child, frozen, fingerprint));
                }
                finally
                {
                    UnityEngine.Object.DestroyImmediate(child);
                    UnityEngine.Object.DestroyImmediate(recipe);
                    UnityEngine.Object.DestroyImmediate(parent);
                }
            }
            var foreignOwner = CloneBaseline();
            baselineType.GetField("ProfileGuid").SetValue(foreignOwner, Guid.NewGuid().ToString("N"));
            Reject("foreign-owner-guid", () => InvokeSubsetContract(binding, profiles[1], foreignOwner, fingerprint));
            Reject("foreign-settings-fingerprint", () => InvokeSubsetContract(binding, profiles[1], frozen, new string('0', 64)));
            var registration = CloneBaseline();
            ((HybridCLRManifest)baselineType.GetField("InitialManifest").GetValue(registration)).HotUpdateAssemblies = Array.Empty<HybridCLRPayload>();
            Reject("outside-global-registration", () => InvokeSubsetContract(validateSet, registration, inputs));
            var metadata = CloneBaseline();
            ((HybridCLRManifest)baselineType.GetField("InitialManifest").GetValue(metadata)).AotMetadata = Array.Empty<HybridCLRPayload>();
            Reject("unregistered-metadata", () => InvokeSubsetContract(validateSet, metadata, inputs));
            RequireSubset(((string[])baselineType.GetField("NativeRequirements").GetValue(frozen)).Length > 0, "The real baseline must actually exercise native requirements.");
            var nativeRequirements = CloneBaseline();
            baselineType.GetField("NativeRequirements").SetValue(nativeRequirements, Array.Empty<string>());
            Reject("unregistered-native-requirements", () => InvokeSubsetContract(validateSet, nativeRequirements, inputs));
            RequireSubset(beforePlayer == SubsetFingerprint(SubsetPlayer) && beforeBaseline == SubsetFingerprint(SubsetBaseline(owner)), "Constraint probes changed a persistent native input.");
            SubsetProfiles(owner);
            Receipt(Run, new { passed = true, UnityVersion = Application.unityVersion, UnityRevision = ExpectedUnityRevision,
                Receiver = owner.Value<string>("receiver"), OwnerSha256 = BoundOwnerSha256,
                Cases = cases, PersistentFilesChanged = false, PlayerFingerprint = beforePlayer, BaselineFingerprint = beforeBaseline });
        }
        public static void PublishProjectSubsetConflict()
        {
            var owner = Owned();
            RequireSubsetRun();
            RequireSubsetEnvironment();
            var native = ReadSubsetNative(owner);
            var profiles = SubsetProfiles(owner);
            var patch = JObject.Parse(File.ReadAllText(Root + "/project-subsets-patch.json"));
            RequireBoundReceipt(owner, patch);
            RequireNewSubsetPath(Root + "/project-subsets-conflict.json");
            var beforePlayer = SubsetFingerprint(SubsetPlayer);
            var beforeBaseline = SubsetFingerprint(SubsetBaseline(owner));
            var manifestPath = profiles[1].generatedAssetDirectory + "/hybridclr/manifest.json";
            var originalManifest = File.ReadAllBytes(manifestPath);
            var manifest = HybridCLRManifest.Parse(Encoding.UTF8.GetString(originalManifest));
            var support = manifest.HotUpdateAssemblies.Single(p => p.Name == SupportName);
            var payloadPath = profiles[1].generatedAssetDirectory + "/" + support.Address;
            var originalPayload = File.ReadAllBytes(payloadPath);
            var originalDescriptor = JObject.FromObject(support);
            try
            {
                byte[] changed;
                using (var module = dnlib.DotNet.ModuleDefMD.Load(originalPayload))
                using (var output = new MemoryStream())
                {
                    var identity = new AssemblyName(module.Assembly.FullName);
                    if (identity.CultureInfo == null) { identity.CultureInfo = System.Globalization.CultureInfo.InvariantCulture; }
                    RequireSubset(identity.FullName == support.Identity, "The real Support identity must be preserved.");
                    module.Mvid = Guid.NewGuid();
                    support.Mvid = module.Mvid.Value.ToString("D");
                    module.Write(output);
                    changed = output.ToArray();
                }
                support.Sha256 = HybridCLRManifest.Hash(changed);
                support.Length = changed.LongLength;
                manifest.Seal();
                manifest.Validate("hclr-sub-b", SubsetCompatibility, Application.unityVersion, true);
                File.WriteAllBytes(payloadPath, changed);
                File.WriteAllText(manifestPath, manifest.ToJson(), new UTF8Encoding(false));
                AssetDatabase.ImportAsset(payloadPath, ImportAssetOptions.ForceSynchronousImport | ImportAssetOptions.ForceUpdate);
                AssetDatabase.ImportAsset(manifestPath, ImportAssetOptions.ForceSynchronousImport | ImportAssetOptions.ForceUpdate);
                var report = ResourceEditorOperations.ExecuteRecipe(profiles[1].recipe);
                RequireSubset(report.Release != null && report.BuildError == null && report.PublicationError == null, "The conflict fixture must be a real published resource release.");
                var published = ReadSubsetRelease(profiles[1], report.Release);
                var actual = published.HotUpdateAssemblies.Single(p => p.Name == SupportName);
                RequireSubset(actual.Identity == support.Identity && actual.Mvid == support.Mvid && actual.Sha256 == support.Sha256 && actual.Mvid != originalDescriptor.Value<string>("Mvid"), "Conflict must be a valid PE with the same name/identity and different MVID/bytes.");
                Receipt("project-subsets-conflict", new { passed = true, UnityVersion = Application.unityVersion, UnityRevision = ExpectedUnityRevision,
                    Receiver = owner.Value<string>("receiver"), OwnerSha256 = BoundOwnerSha256,
                    ProducerRun = Run, Baseline = SubsetBaseline(owner), CompatibilityId = SubsetCompatibility,
                    PlayerFingerprint = beforePlayer, BaselineFingerprint = beforeBaseline, ReleaseB = SubsetReleaseIdentity(report.Release, published),
                    OriginalSupport = originalDescriptor, ChangedSupport = actual, NativeVariant = native?.Value<string>("Variant") ?? "OfficialSdkLocal",
                    NativeSourceManifestSha256 = owner.Value<string>("nativeSourceManifestSha256"), SubsetConsumerVersion = 2, FixtureOnly = true });
            }
            finally
            {
                File.WriteAllBytes(payloadPath, originalPayload);
                File.WriteAllBytes(manifestPath, originalManifest);
                AssetDatabase.ImportAsset(payloadPath, ImportAssetOptions.ForceSynchronousImport | ImportAssetOptions.ForceUpdate);
                AssetDatabase.ImportAsset(manifestPath, ImportAssetOptions.ForceSynchronousImport | ImportAssetOptions.ForceUpdate);
            }
            RequireSubset(originalPayload.SequenceEqual(File.ReadAllBytes(payloadPath)) && originalManifest.SequenceEqual(File.ReadAllBytes(manifestPath)) &&
                beforePlayer == SubsetFingerprint(SubsetPlayer) && beforeBaseline == SubsetFingerprint(SubsetBaseline(owner)), "Conflict fixture failed to restore original source/baseline bytes.");
            Receipt(Run, new { passed = true, SourceRestored = true, PlayerFingerprint = beforePlayer, BaselineFingerprint = beforeBaseline });
        }

        private static void RejectSubsetLinks(string path)
        {
            for (var cursor = Path.GetFullPath(path); cursor != null; cursor = Path.GetDirectoryName(cursor))
            {
                FileAttributes attributes;
                try { attributes = File.GetAttributes(cursor); }
                catch (FileNotFoundException) { continue; }
                catch (DirectoryNotFoundException) { continue; }
                if ((attributes & FileAttributes.ReparsePoint) != 0)
                {
                    throw new InvalidDataException("Linked project-subset inputs/outputs are not supported: " + cursor);
                }
            }
        }

        private static HybridCLRBuildProfile[] SubsetProfiles(JObject owner)
        {
            var profiles = new[] { "A", "B" }.Select(s => AssetDatabase.LoadAssetAtPath<HybridCLRBuildProfile>(SubsetAssets + "/Profile" + s + ".asset")).ToArray();
            for (var i = 0; i < profiles.Length; i++)

                var project = recipe.project;

                    project.projectId == (i == 0 ? "hclr-sub-a" : "hclr-sub-b") && profile.development && profile.generatedAssetDirectory == generated &&
                    SameSubsetPath(profile.baselineDirectory, SubsetBaseline(owner)) && profile.hotUpdateAssemblies != null &&
                    profile.hotUpdateAssemblies.SequenceEqual(new[] { i == 0 ? SupportName : EntryName }) && recipe.target == BuildTarget.StandaloneWindows64 &&
                    recipe.action == ResourceBuildAction.BuildAndPublish && recipe.appContentCompatibilityId == SubsetCompatibility && !recipe.generateReferences &&
                    !EditorUtility.IsDirty(profile) && !EditorUtility.IsDirty(recipe) && !EditorUtility.IsDirty(project), "The saved subset profile/recipe/project contract changed.");
                RequireSubset(project.sourceDirectories.Count == 1 && project.sourceDirectories[0].guid == AssetDatabase.AssetPathToGUID(generated) &&
                    !string.IsNullOrEmpty(project.sourceDirectories[0].guid) && string.IsNullOrEmpty(project.sourceDirectories[0].addressPrefix) &&
                    project.groups.Count == 1 && project.groups[0].id == "code" && project.groups[0].selections.Count == 1 &&
                    project.groups[0].selections[0].guid == AssetDatabase.AssetPathToGUID(generated + "/hybridclr") &&
                    !string.IsNullOrEmpty(project.groups[0].selections[0].guid) && string.IsNullOrEmpty(project.groups[0].selections[0].address) &&
                    project.groups[0].selections[0].extensions.SequenceEqual(new[] { ".bytes", ".json" }), "Keep the explicit generated source + hybridclr code selection mapping.");
            }
            RequireSubset(!profiles[0].baseProfile && profiles[1].baseProfile == profiles[0] && EditorBuildSettings.scenes.Length == 1 &&
                EditorBuildSettings.scenes[0].enabled && EditorBuildSettings.scenes[0].path == SubsetScene && File.Exists(SubsetScene), "Only the new Probe scene and one-hop B -> A base binding are allowed.");

        }

        private static HybridCLRManifest ReadSubsetRelease(HybridCLRBuildProfile profile, ResourceRelease release)
        {
            RequireSubset(release != null, "Missing actual project-subset release.");
            var source = HybridCLRManifest.Parse(File.ReadAllText(profile.generatedAssetDirectory + "/hybridclr/manifest.json"));
            source.Validate(profile.recipe.project.projectId, SubsetCompatibility, Application.unityVersion, true);
            var snapshot = ResourceLocalSnapshot.Open(release.DirectoryPath, source.ProjectId, release.Catalog.CatalogDigest, "StandaloneWindows64", SubsetCompatibility);
            var bundles = new Dictionary<string, AssetBundle>(StringComparer.Ordinal);
            byte[] Read(string address)
            {
                var asset = snapshot.Document.Assets.Single(a => a.Address == address);
                RequireSubset(asset.AssetType == typeof(TextAsset).FullName, "The code-only fixture must contain TextAssets.");
                if (!bundles.TryGetValue(asset.BundleId, out var loaded))
                {
                    var bundle = snapshot.Document.Bundles.Single(b => b.LogicalId == asset.BundleId);
                    RequireSubset(bundle.Dependencies.Count == 0, "The code-only fixture must not acquire unrelated native Bundle dependencies.");
                    loaded = AssetBundle.LoadFromFile(Path.Combine(release.DirectoryPath, bundle.RelativeFile));
                    RequireSubset(loaded, "Cannot read the actual published code Bundle.");
                    bundles.Add(asset.BundleId, loaded);
                }
                var text = loaded.LoadAsset<TextAsset>(asset.LoadName);
                RequireSubset(text, "Missing actual published code TextAsset: " + address);
                return text.bytes;
            }
            try
            {
                var manifest = HybridCLRManifest.Parse(Encoding.UTF8.GetString(Read(HybridCLRManifest.ManifestPath)));
                RequireSubset(manifest.ToJson() == source.ToJson(), "The actual published manifest differs from its generated source manifest.");
                foreach (var payload in manifest.HotUpdateAssemblies.Concat(manifest.AotMetadata))
                {
                    HybridCLRManifest.ValidatePayload(payload, Read(payload.Address));
                }

                return manifest;
            }
            finally
            {
                foreach (var bundle in bundles.Values)
                {
                    bundle.Unload(true);
                }
            }
        }

        private static string SubsetFingerprint(string directory)
        {
            directory = Path.GetFullPath(directory).TrimEnd('/', '\\');
            RequireSubset(SameSubsetPath(directory, SubsetPlayer) || SameSubsetPath(directory, SubsetBaseline(Owned())), "Fingerprint only this round's explicit Player or baseline.");
            RejectSubsetLinks(directory);
            if (!Directory.Exists(directory))
            {
                throw new DirectoryNotFoundException(directory);
            }

            var files = new List<string>();
            var pending = new Stack<string>(new[] { directory });
            while (pending.Count != 0)
            {
                foreach (var path in Directory.GetFileSystemEntries(pending.Pop()))
                {
                    RejectSubsetLinks(path);
                    if (Directory.Exists(path))
                    {
                        pending.Push(path);
                    }
                    else
                    {
                        files.Add(path);
                    }
                }
            }

            RequireSubset(files.Count != 0, "Cannot fingerprint an empty acceptance output.");
            return HybridCLRManifest.Hash(Encoding.UTF8.GetBytes(string.Join("\n", files.OrderBy(p => p, StringComparer.Ordinal)
                .Select(p => p.Substring(directory.Length + 1).Replace('\\', '/') + ":" + Sha(p)))));
        }

            .Where(s => new[] { "Il2CppDef", "LinkXml", "EmptyMethodBridge", "StripAot", "FullMethodBridge", "Player" }.Contains(s.Name)).Select(s => s.Name).ToArray();

        public static void PrepareDevelopmentNegatives()
        {
            var owner = Owned();
            var baseline = JObject.Parse(File.ReadAllText(Root + "/" + owner.Value<string>("baseReceiptStem") + ".json"));
            RequireBoundReceipt(owner, baseline);
            var profile = Profile();

            RequireSubset(baseline.Value<bool>("passed") && baseline.Value<bool>("Development") &&
                priorPlayer == baseline.Value<string>("PlayerFingerprint") &&
                priorBaseline == baseline.Value<string>("NativeBaselineSha256"), "The original frozen Development base must remain intact.");
            EditorUserBuildSettings.development = true;
            profile.development = true;
            profile.recipe.appContentCompatibilityId = owner.Value<string>("compatibilityId");

            EditorUtility.SetDirty(profile);
            EditorUtility.SetDirty(profile.recipe);

            AssetDatabase.SaveAssetIfDirty(profile.recipe);
            var report = HybridCLRBuildPipeline.PublishPatch(profile);
            WriteReport(report);
            RequireSubset(SubsetNativeStages(report).Length == 0 && priorPlayer == PlayerFingerprint() &&
                priorBaseline == Sha(baselinePath + "/baseline.json"), "Preparing negative working payloads must not rebuild or modify the frozen base.");
            Receipt(Run + "-frozen", new
            {
                passed = true,
                restoredDevelopmentWorkingInputs = true,
                nativeStages = SubsetNativeStages(report),
                PlayerFingerprint = priorPlayer,
                BaselineSha256 = priorBaseline,
                report.Release.Catalog.CatalogDigest,
                report.Manifest.CodePackageId
            });
        }
        public static void RejectNewRequirement()
        {
            Owned();
            var profile = Profile();
            var priorPlayer = PlayerFingerprint();
            var priorBaseline = Sha(profile.baselineDirectory + "/baseline.json");
            var report = HybridCLRBuildPipeline.PublishPatch(profile);
            var rejected = !report.HasSucceeded && report.Stages.Last().Name == "PatchCompatibility" &&
                report.Stages.Last().Error != null && report.Stages.Last().Error.Message.Contains("New AOT generic/native bridge requirements");
            if (!rejected || report.Release != null || priorPlayer != PlayerFingerprint() || priorBaseline != Sha(profile.baselineDirectory + "/baseline.json"))
            {
                throw new InvalidOperationException("The new native requirement was not rejected against an unchanged frozen base: " + report.Summary);
            }

            Receipt(Run, new
            {
                passed = true,
                newRequirementRejected = true,
                unchangedPlayer = priorPlayer,
                unchangedBaseline = priorBaseline,
                report.Summary
            });
        }

        private static void Sign(ResourceRelease release, string destination, long sequence)
        {
            using var certificate = new X509Certificate2(Root + "/TestCredentials/local-only.pfx", "hclr-local-fixture", X509KeyStorageFlags.EphemeralKeySet);
            using var key = certificate.GetRSAPrivateKey();
            var directory = Root + "/" + destination;
            if (Directory.Exists(directory))
            {
                throw new IOException("Do not replace a signed server fixture.");
            }

            Directory.CreateDirectory(directory + "/" + release.Catalog.CatalogDigest);
            var envelope = ResourceReleaseSigner.Create(release, "hybridclr-test", sequence, "hclr-test", key, DateTimeOffset.UtcNow.AddDays(1));
            File.WriteAllBytes(directory + "/release.json", envelope);
            foreach (var path in Directory.GetFiles(release.DirectoryPath, "*", SearchOption.AllDirectories))
            {
                var relative = path.Substring(release.DirectoryPath.Length + 1);
                var target = directory + "/" + release.Catalog.CatalogDigest + "/" + relative;
                Directory.CreateDirectory(Path.GetDirectoryName(target));
                File.Copy(path, target, false);
            }
        }
