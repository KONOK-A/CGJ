using System;
using System.Collections;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Reflection;
using Newtonsoft.Json;
using Newtonsoft.Json.Linq;
using UnityEngine;
using YLFramework.Integrations.HybridCLR;
using YLFramework.Samples.HybridCLR;
using YLFramework.Resource;

namespace YLFramework.Tests.HybridCLRReception
{
    // Real acceptance consumer, not a dependency shim. Its explicit Development-only TLS pin uses the existing test friend.
    public sealed class HybridCLRReceptionRunner : MonoBehaviour
    {
        private const string ExpectedUnityVersion = "__YL_HYBRIDCLR_EXPECTED_UNITY_VERSION__";
        private const string ExpectedUnityRevision = "__YL_HYBRIDCLR_EXPECTED_UNITY_REVISION__";
        private const string ExpectedRoot = "__YL_HYBRIDCLR_EXPECTED_ROOT__";
        private const string BoundOwnerSha256 = "__YL_HYBRIDCLR_OWNER_SHA256__";
        [global::UnityEngine.Serialization.FormerlySerializedAs("ProjectId")]
        public string projectId;
        [global::UnityEngine.Serialization.FormerlySerializedAs("CompatibilityId")]
        public string compatibilityId;
        private JObject _evidence;
        private string _root;
        private string _action;
        private string _run;
        private HybridCLRExample _sample;

        private string Argument(string name)
        {
            var args = Environment.GetCommandLineArgs();
            var index = Array.IndexOf(args, name);
            if (index < 0 || index + 1 == args.Length || index != Array.LastIndexOf(args, name))
            {
                throw new ArgumentException("One explicit " + name + " is required.");
            }

            return args[index + 1];
        }
        private IEnumerator Start()
        {
            Application.runInBackground = true;
            var stack = new Stack<IEnumerator>();
            stack.Push(Run());
            Exception failure = null;
            while (stack.Count != 0)
            {
                bool next = false;
                object current = null;
                try
                {
                    next = stack.Peek().MoveNext();
                    if (next)
                    {
                        current = stack.Peek().Current;
                    }
                }
                catch (Exception error)
                {
                    failure = error;
                }
                if (failure != null)
                {
                    break;
                }

                if (!next)
                {
                    (stack.Pop() as IDisposable)?.Dispose();
                    continue;
                }
                if (current is IEnumerator nested && !(current is CustomYieldInstruction))
                {
                    stack.Push(nested);
                    continue;
                }
                yield return current;
            }
            foreach (var remaining in stack)
            {
                try
                {
                    (remaining as IDisposable)?.Dispose();
                }
                catch (Exception error)
                {
                    Debug.LogException(error);
                }
            }

            _evidence = _evidence ?? new JObject();
            _evidence["passed"] = failure == null;
            if (failure != null)
            {
                _evidence["error"] = failure.ToString();
                Debug.LogException(failure);
            }
            try
            {
                var output = _root + "/" + _run + ".json";
                using var file = new FileStream(output, FileMode.CreateNew, FileAccess.Write, FileShare.Read);
                var bytes = System.Text.Encoding.UTF8.GetBytes(_evidence.ToString(Formatting.Indented));
                file.Write(bytes, 0, bytes.Length);
                Debug.Log("HYBRIDCLR_ACCEPTANCE " + _evidence.ToString(Formatting.None));
            }
            catch (Exception error)
            {
                Debug.LogException(error);
                failure = failure ?? error;
            }
            Application.Quit(failure == null ? 0 : 21);
        }
        private IEnumerator Run()
        {
            _root = Path.GetFullPath(Argument("-hclr-root"));
            _action = Argument("-hclr-action");
            _run = Argument("-hclr-run");
            if (_action != "V1" && _action != "Update" && _action != "LocalUpdate" && _action != "V2" && _action != "BadPayload" && _action != "BadMetadata" && _action != "BadSignature" && _action != "Parameters")
            {
                throw new ArgumentException("Unknown explicit acceptance action.");
            }

            if (Application.platform != RuntimePlatform.WindowsPlayer || IntPtr.Size != 8 || Application.unityVersion != ExpectedUnityVersion ||
                !Path.GetFullPath(_root).Equals(Path.GetFullPath(ExpectedRoot), StringComparison.OrdinalIgnoreCase))
            {
                throw new InvalidOperationException("The acceptance receiver requires its actual Windows64 IL2CPP Player.");
            }

            var baseline = JObject.Parse(File.ReadAllText(Argument("-hclr-base-receipt")));
            var ownerPath = Path.Combine(_root, "owner.json");
            var owner = JObject.Parse(File.ReadAllText(ownerPath));
            if (BoundOwnerSha256.Length != 0)
            {
#if !ENABLE_IL2CPP
                throw new PlatformNotSupportedException("This owned acceptance Player must use IL2CPP.");
#endif
                Require(SystemInfo.graphicsDeviceType != UnityEngine.Rendering.GraphicsDeviceType.Null, "A real graphics device is required; headless acceptance is not allowed.");
                Require(HybridCLRManifest.Hash(File.ReadAllBytes(ownerPath)) == BoundOwnerSha256 && !owner.Value<bool>("simulation") &&
                    baseline.Value<string>("UnityVersion") == ExpectedUnityVersion && baseline.Value<string>("UnityRevision") == ExpectedUnityRevision &&
                    baseline.Value<string>("Receiver") == owner.Value<string>("receiver") && baseline.Value<string>("OwnerSha256") == BoundOwnerSha256 &&
                    Path.GetFullPath(baseline.Value<string>("Baseline")) == Path.GetFullPath(owner.Value<string>(Debug.isDebugBuild ? "baseline" : "releaseBaseline")) &&
                    Path.GetFullPath(Path.GetDirectoryName(Application.dataPath)) == Path.GetFullPath(owner.Value<string>(Debug.isDebugBuild ? "playerRoot" : "releasePlayerRoot")),
                    "Cross-engine, foreign-owner or unknown Player/base receipt.");
            }
            if (baseline.Value<string>("ProjectId") != projectId || baseline.Value<string>("CompatibilityId") != compatibilityId)
            {
                throw new InvalidDataException("Player/base receipt mismatch.");
            }

            Require(baseline.Value<bool>("Development") == Debug.isDebugBuild, "The actual Player Development flag differs from its base receipt.");
            _evidence = new JObject { ["action"] = _action, ["unityVersion"] = Application.unityVersion, ["projectId"] = projectId, ["simulation"] = false };
            var audio = AudioSettings.GetConfiguration();
            _evidence["graphicsDevice"] = SystemInfo.graphicsDeviceType.ToString();
            _evidence["graphicsName"] = SystemInfo.graphicsDeviceName;
            _evidence["width"] = Screen.width;
            _evidence["height"] = Screen.height;
            _evidence["audioSampleRate"] = audio.sampleRate;
            _evidence["audioDspBufferSize"] = audio.dspBufferSize;
            _evidence["nativeVariant"] = baseline.Value<string>("NativeVariant");
            _evidence["nativeSourceManifestSha256"] = baseline.Value<string>("NativeSourceManifestSha256");
            var snapshot = ResourceLocalSnapshot.OpenBuiltIn(projectId, baseline.Value<string>("CatalogDigest"), "StandaloneWindows64", compatibilityId);
            ResourceRemoteSource remote = null;
            if (_action == "Update" || _action == "BadPayload" || _action == "BadSignature")
            {
                var trust = JObject.Parse(File.ReadAllText(Argument("-hclr-trust")));
                remote = new ResourceRemoteSource(trust.Value<string>("endpoint"), "hybridclr-test", new[] { new ResourceSigningKey("hclr-test", trust.Value<string>("modulus"), trust.Value<string>("exponent")) },
                    maximumFileBytes: 128L * 1024 * 1024, maximumDownloadBytes: 512L * 1024 * 1024, timeoutSeconds: 60, attempts: 1);
#if UNITY_EDITOR || (DEVELOPMENT_BUILD && YL_RESOURCE_ACCEPTANCE_TESTS)
                remote.TrustLoopbackCertificate(Convert.FromBase64String(trust.Value<string>("certificate")));
#else
                throw new PlatformNotSupportedException("Isolated loopback trust is deliberately unavailable in ordinary Release Players.");
#endif
            }
            var cacheRoot = owner.Value<string>("runtimeCacheRoot");
            if (string.IsNullOrEmpty(cacheRoot) || !Path.IsPathRooted(cacheRoot))
            {
                throw new InvalidDataException("An explicit owned short runtime cache root is required.");
            }

            var cache = cacheRoot + (_action == "BadMetadata" ? "/bad-metadata" : _action == "BadPayload" ? "/bad" : _action == "BadSignature" ? "/bad-signature" : Debug.isDebugBuild ? "/normal" : "/release");
            var store = new ArtifactStore(cache, new[] { new ResourceProjectMount(projectId, "StandaloneWindows64", compatibilityId, snapshot, remote: remote) });
            var config = new ResourceConfiguration(store);
            if (_action == "BadMetadata")
            {
                var candidate = JObject.Parse(File.ReadAllText(Argument("-hclr-candidate")));
                var local = ResourceLocalSnapshot.Open(candidate.Value<string>("DirectoryPath"), projectId, candidate.Value<string>("CatalogDigest"), "StandaloneWindows64", compatibilityId);
                var prepared = store.Prepare(store.Check(store.Mount(), new[] { local }), System.Threading.CancellationToken.None);
                var startup = new ResourceStartup(config);
                using (var init = startup.StartAsync())
                {
                    yield return init;
                    init.TakeResult();
                }
                using (var activate = startup.Resources.ActivatePreparedUpdateAsync(prepared))
                {
                    yield return activate;
                    Require(activate.TakeResult(), "The local negative metadata fixture did not activate.");
                }
                using var loader = new HybridCLRLoader(startup.Resources, compatibilityId);
                var failed = loader.LoadAsync(new ResourceAddress(projectId, HybridCLRManifest.ManifestPath));
                yield return failed;
                Require(failed.Exception != null && failed.RequiresRestart && failed.Stage == "Metadata" && failed.AssemblyName == "zzAbsentAot" && failed.NativeErrorCode.HasValue && failed.NativeErrorCode.Value != 0,
                    "The real native metadata rejection did not retain partial-load restart semantics and its exact error code.");
                using (var another = new HybridCLRLoader(startup.Resources, compatibilityId))
                {
                    var blocked = another.LoadAsync(new ResourceAddress(projectId, HybridCLRManifest.ManifestPath));
                    yield return blocked;
                    Require(blocked.Exception != null && blocked.RequiresRestart, "A new loader improperly reset the native partial-load requirement.");
                }
                _evidence["partialMetadataRestartRequired"] = true;
                _evidence["nativeErrorCode"] = failed.NativeErrorCode.Value;
                _evidence["assembly"] = failed.AssemblyName;
                loader.Dispose();
                using var stop = startup.StopAsync();
                yield return stop;
                stop.TakeResult();
                yield break;
            }
            if (_action == "BadSignature")
            {
                var startup = new ResourceStartup(config, bindCore: true);
                using (var init = startup.StartAsync())
                {
                    yield return init;
                    init.TakeResult();
                }
                using var check = startup.Resources.CheckRemoteUpdatesAsync();
                yield return check;
                Require(IsSignatureRejection(check.Error), "Only the exact production signature rejection is a successful authentication negative; network/catalog failures are not substitutes.");
                _evidence["signatureRejectedBeforeNative"] = true;
                _evidence["resourceError"] = check.Error.Kind.ToString();
                _evidence["signatureFailureMessage"] = check.Error.InnerException.Message;
                using var stop = startup.StopAsync();
                yield return stop;
                stop.TakeResult();
                Require(ResourceSystem.Current == null, "Signature rejection must release its Resource/Core binding.");
                _evidence["stopped"] = true;
                yield break;
            }
            if (_action == "BadPayload")
            {
                var startup = new ResourceStartup(config, bindCore: true);
                using (var init = startup.StartAsync())
                {
                    yield return init;
                    init.TakeResult();
                }
                using var check = startup.Resources.CheckRemoteUpdatesAsync();
                yield return check;
                using var prepare = startup.Resources.PrepareUpdateAsync(check.TakeResult());
                yield return prepare;
                using var activate = startup.Resources.ActivatePreparedUpdateAsync(prepare.TakeResult());
                yield return activate;
                Require(activate.TakeResult(), "Bad fixture did not activate before code loading.");
                using var loader = new HybridCLRLoader(startup.Resources, compatibilityId);
                var bad = loader.LoadAsync(new ResourceAddress(projectId, HybridCLRManifest.ManifestPath));
                yield return bad;
                Require(bad.Exception != null && !bad.RequiresRestart && bad.Stage == "Payloads", "Corrupt code was not rejected before native loading.");
                _evidence["badPayloadRejectedBeforeNative"] = true;
                _evidence["exception"] = bad.Exception.ToString();
                loader.Dispose();
                using var stop = startup.StopAsync();
                yield return stop;
                stop.TakeResult();
                yield break;
            }
            if (_action == "Parameters")
            {
                Require(!Debug.isDebugBuild && baseline["ParameterOracle"] != null && owner["nativeSourceManifest"] != null &&
                    baseline.Value<string>("NativeSourceManifestSha256") == owner.Value<string>("nativeSourceManifestSha256"), "The exact ordinary Release native experiment/oracle is required.");
                var startup = new ResourceStartup(config);
                using (var init = startup.StartAsync())
                {
                    yield return init;
                    init.TakeResult();
                }
                using var loader = new HybridCLRLoader(startup.Resources, compatibilityId);
                var request = loader.LoadAsync(new ResourceAddress(projectId, HybridCLRManifest.ManifestPath));
                yield return request;
                Require(request.Exception == null && !request.Result.IsSimulation, "Real metadata and interpreter assemblies must load before reflection.");
                var model = request.Result.Assemblies[HybridCLRExample.EntryAssembly].GetType("YLFramework.Samples.HybridCLR.HotUpdate.ExampleMessage", true);
                var array = model.MakeArrayType();
                var cases = new JArray();
                _evidence["parameterCases"] = cases;
                _evidence["parameterOracle"] = baseline["ParameterOracle"].DeepClone();
                foreach (var type in new[] { array, typeof(int[]), typeof(string[]) })
                {
                    var methods = type.GetMethods(BindingFlags.Instance | BindingFlags.Public | BindingFlags.NonPublic | BindingFlags.DeclaredOnly)
                        .Where(m => m.Name.EndsWith(".RemoveAt", StringComparison.Ordinal)).ToArray();
                    Require(methods.Length > 0, "Actual native array interface methods are missing.");
                    var oracle = baseline["ParameterOracle"];
                    foreach (var method in methods)
                    {
                        Debug.Log("B08_PARAMETER_BEGIN " + type.FullName + " " + method.Name + " " + method.MetadataToken);
                        var parameters = method.GetParameters();
                        Require(parameters.Length == 1 && parameters[0].Position == 0 && parameters[0].Name == oracle.Value<string>("Name") &&
                            parameters[0].ParameterType.FullName == oracle.Value<string>("Type") && (uint)parameters[0].MetadataToken == oracle.Value<uint>("Token") &&
                            (uint)method.MetadataToken == oracle.Value<uint>("MethodToken"), "Every native synthetic array parameter must belong to the final AOT definition.");
                    }
                    cases.Add(new JObject { ["Case"] = type == array ? "hot-array-aot-definition" : type == typeof(int[]) ? "aot-value-array" : "aot-reference-array",
                        ["Name"] = oracle.Value<string>("Name"), ["Type"] = oracle.Value<string>("Type"), ["Token"] = oracle.Value<uint>("Token"),
                        ["MethodToken"] = oracle.Value<uint>("MethodToken"), ["DeclaringType"] = type.FullName, ["MethodVariants"] = methods.Length, ["Passed"] = true });
                }
                var contains = array.GetMethods(BindingFlags.Instance | BindingFlags.Public | BindingFlags.NonPublic | BindingFlags.DeclaredOnly)
                    .Where(m => m.Name.EndsWith(".Contains", StringComparison.Ordinal)).Select(m => m.GetParameters()).ToArray();
                Require(contains.Length > 0 && contains.All(p => p.Length == 1 && p[0].Name == "item") && contains.Any(p => p[0].ParameterType == model),
                    "Generic array inflation must keep its actual interpreter element type and all covariant variants valid.");
                cases.Add(new JObject { ["Case"] = "generic-array-parameter", ["Passed"] = true, ["Type"] = model.FullName, ["MethodVariants"] = contains.Length });
                var ordinary = model.GetProperty("Text").SetMethod.GetParameters();
                Require(ordinary.Length == 1 && ordinary[0].ParameterType == typeof(string) && ordinary[0].Name == "value", "Ordinary interpreter method parameter was redirected incorrectly.");
                cases.Add(new JObject { ["Case"] = "ordinary-interpreter-method", ["Passed"] = true });
                var pseudo = array.GetMethod("Get", BindingFlags.Public | BindingFlags.Instance | BindingFlags.DeclaredOnly);
                var pseudoParameters = pseudo.GetParameters();
                Require(pseudo.MetadataToken == 0 && pseudoParameters.Length == 1 && pseudoParameters[0].ParameterType == typeof(int), "Array pseudo-method without a definition must remain valid.");
                cases.Add(new JObject { ["Case"] = "array-pseudo-method-null-handle", ["Passed"] = true });
                _evidence["parameterCases"] = cases;
                _evidence["parameterOracle"] = baseline["ParameterOracle"].DeepClone();
                loader.Dispose();
                using var stop = startup.StopAsync();
                yield return stop;
                stop.TakeResult();
                Require(ResourceSystem.Current == null, "Narrow parameter probe leaked the resource binding.");
                _evidence["stopped"] = true;
                yield break;
            }
            _sample = new HybridCLRExample();
            yield return _sample.Run(config, projectId, compatibilityId);
            Require(!_sample.ModuleRequest.Code.IsSimulation, "Editor Simulation is not native acceptance.");
            var expectedVersion = _action == "V2" ? "v2:" : "v1:";
            Require(HybridCLRExampleState.Message != null && HybridCLRExampleState.Message.StartsWith(expectedVersion, StringComparison.Ordinal) &&
                HybridCLRExampleState.Message.Contains("YL Resource configuration") && _sample.ModuleRequest.Module.State == YLFramework.Core.ModuleState.Running,
                "Actual interpreted version or Core configuration/Module state mismatch: " + _sample.Result);
            Require(_sample.Result.Contains(expectedVersion.TrimEnd(':') + "-async") && _sample.PrefabMarker == expectedVersion.TrimEnd(':') + "-prefab", "Actual hot async continuation or prefab MonoBehaviour result mismatch.");
            _evidence["message"] = _sample.Result;
            _evidence["codePackageId"] = _sample.ModuleRequest.Code.CodePackageId;
            _evidence["catalogDigest"] = _sample.Startup.Resources.GetActiveCatalogDigest(projectId);
            _evidence["entryIdentity"] = _sample.ModuleRequest.Code.Assemblies[HybridCLRExample.EntryAssembly].FullName;
            _evidence["mvidEvidence"] = "Verified from PE bytes before native load; CLR module GUID reflection is not supported by this IL2CPP.";
            var repeat = _sample.Loader.LoadAsync(new ResourceAddress(projectId, HybridCLRManifest.ManifestPath));
            yield return repeat;
            Require(ReferenceEquals(repeat.Result, _sample.ModuleRequest.Code), "Repeated code request was not idempotent.");
            using (var wrongBase = new HybridCLRLoader(_sample.Startup.Resources, "wrong-base"))
            {
                var wrong = wrongBase.LoadAsync(new ResourceAddress(projectId, HybridCLRManifest.ManifestPath));
                yield return wrong;
                Require(wrong.Exception != null && wrong.Stage == "Manifest" && !wrong.RequiresRestart, "Wrong base was not rejected before native load.");
                _evidence["wrongBaseRejected"] = true;
            }
            if (_action == "Update" || _action == "LocalUpdate")
            {
                UpdatePlan plan;
                if (_action == "LocalUpdate")
                {
                    Require(!Debug.isDebugBuild && owner["nativeSourceManifest"] != null, "Owned ordinary Release local update only.");
                    var candidatePath = Argument("-hclr-candidate");
                    var candidate = JObject.Parse(File.ReadAllText(candidatePath));
                    Require(string.Equals(Path.GetFullPath(candidatePath), Path.GetFullPath(Path.Combine(_root, "release-patch-receipt.json")), StringComparison.OrdinalIgnoreCase) &&
                        candidate.Value<bool>("passed") && !candidate.Value<bool>("Development") &&
                        candidate.Value<string>("OwnerSha256") == BoundOwnerSha256 && candidate.Value<string>("PlayerFingerprint") == baseline.Value<string>("PlayerFingerprint") &&
                        candidate.Value<string>("BaselineSha256") == baseline.Value<string>("NativeBaselineSha256") &&
                        candidate.Value<string>("NativeSourceManifestSha256") == owner.Value<string>("nativeSourceManifestSha256"), "Local candidate does not belong to this frozen Release base.");
                    var directory = Path.GetFullPath(candidate.Value<string>("DirectoryPath"));
                    Require(directory.StartsWith(Path.GetFullPath(owner.Value<string>("project")) + Path.DirectorySeparatorChar, StringComparison.OrdinalIgnoreCase), "Candidate is outside the exact owned publisher project.");
                    var local = ResourceLocalSnapshot.Open(directory, projectId, candidate.Value<string>("CatalogDigest"), "StandaloneWindows64", compatibilityId);
                    plan = store.Check(store.Mount(), new[] { local });
                    _evidence["updateTransport"] = "Explicit owned published local snapshot";
                    _evidence["remoteTlsTested"] = false;
                }
                else
                {
                    using var check = _sample.Startup.Resources.CheckRemoteUpdatesAsync();
                    yield return check;
                    plan = check.TakeResult();
                }
                using var prepare = _sample.Startup.Resources.PrepareUpdateAsync(plan);
                yield return prepare;
                var prepared = prepare.TakeResult();
                _evidence["checkAndPrepareWhileRunning"] = true;
                using (var busy = _sample.Startup.Resources.ActivatePreparedUpdateAsync(prepared))
                {
                    yield return busy;
                    Require(busy.Error != null && busy.Error.Kind == ResourceError.Busy, "Live manifest lease did not block Activate.");
                    _evidence["activateBusy"] = true;
                }
                Require(HybridCLRExampleState.Message.StartsWith("v1:", StringComparison.Ordinal), "Busy changed the running code.");
                _sample.CloseCode();
                var committed = false;
                for (var frame = 0; frame < 240 && !committed; frame++)
                {
                    using var activated = _sample.Startup.Resources.ActivatePreparedUpdateAsync(prepared);
                    yield return activated;
                    if (activated.Status == ResourceRequestStatus.Succeeded)
                    {
                        committed = activated.TakeResult();
                    }
                    else
                    {
                        Require(activated.Error != null && activated.Error.Kind == ResourceError.Busy, "Activation failed instead of draining owned native instance destruction.");
                    }

                    if (!committed)
                    {
                        yield return null;
                    }
                }
                Require(committed, "Closed code/instance responsibilities did not drain before activation.");
                using (var another = new HybridCLRLoader(_sample.Startup.Resources, compatibilityId))
                {
                    var conflict = another.LoadAsync(new ResourceAddress(projectId, HybridCLRManifest.ManifestPath));
                    yield return conflict;
                    Require(conflict.Exception != null && conflict.RequiresRestart, "Dispose/new loader incorrectly allowed same-process code replacement.");
                    _evidence["sameProcessReplacementRejected"] = true;
                }
                _evidence["activatedCatalogDigest"] = _sample.Startup.Resources.GetActiveCatalogDigest(projectId);
            }
            using var shutdown = _sample.CloseAsync();
            yield return shutdown;
            shutdown.TakeResult();
            Require(ResourceSystem.Current == null, "Owned ResourceStartup did not close cleanly.");
        }
        private static bool IsSignatureRejection(Exception error) => error is ResourceException failure && failure.Kind == ResourceError.UpdateFailed &&
            failure.InnerException is InvalidDataException invalid && invalid.Message == "Release signature rejected.";

        private static void Require(bool condition, string error)
        {
            if (!condition)
            {
                throw new InvalidOperationException(error);
            }
        }
    }
}
